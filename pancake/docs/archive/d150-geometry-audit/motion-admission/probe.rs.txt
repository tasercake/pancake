//! Trace actual product estimates on a small frozen D150 witness set, on CPU.

use std::path::{Path, PathBuf};

use pancake::image_io::decode;
use pancake::registration::{Frame, Model, Thumbnail, estimate_pair, thumbnail_with_scale};
use serde_json::Value;

/// Keep the exact first-frame name, native geometry and device-generated proxy.
struct Input {
    /// Original basename in frozen order.
    name: String,
    /// Packed native source dimensions.
    size: [usize; 2],
    /// Shared registration luma proxy.
    proxy: Thumbnail,
}

/// Process the anchor and both four-frame windows on the production CPU code.
/// # Errors
/// Returns launch, decode, runtime or output errors.
fn main() -> Result<(), String> {
    let mut arguments = std::env::args_os().skip(1).map(PathBuf::from);
    let destination = arguments.next().ok_or("missing diagnostic output path")?;
    let paths = arguments.collect::<Vec<_>>();
    if paths.len() != 9 {
        return Err("expected an anchor and two four-source windows".to_owned());
    }
    let device = cubecl::Device::cpu().map_err(|error| error.to_string())?;
    let inputs = paths
        .iter()
        .map(|path| prepare(&device, path))
        .collect::<Result<Vec<_>, _>>()?;
    let [anchor, first_a, first_b, first_c, first_d, second_a, second_b, second_c, second_d] =
        inputs.as_slice()
    else {
        return Err("diagnostic image count changed".to_owned());
    };
    if inputs.iter().any(|input| input.size != anchor.size) {
        return Err("D150 native source dimensions disagree".to_owned());
    }
    let mut estimates = Vec::new();
    for window in [[first_a, first_b, first_c, first_d], [second_a, second_b, second_c, second_d]] {
        for input in window {
            compare_both(&device, "anchor", anchor, input, &mut estimates)?;
        }
        for pair in window.windows(2) {
            let &[before, after] = pair else { return Err("probe window has an invalid pair".to_owned()); };
            compare_both(&device, "neighbor", before, after, &mut estimates)?;
        }
    }
    let result = serde_json::json!({
        "runtime": "cpu", "scale_factor": 6, "max_scale_step": 0.05, "max_precision": false,
        "source_order": inputs.iter().map(|input| &input.name).collect::<Vec<_>>(),
        "first_canvas": anchor.size, "estimates": estimates,
    });
    let output = std::fs::OpenOptions::new().write(true).create_new(true)
        .open(destination).map_err(|error| error.to_string())?;
    serde_json::to_writer_pretty(output, &result).map_err(|error| error.to_string())
}

/// Decode and thumbnail through the same public application entries.
/// # Errors
/// Returns image, path or CPU preparation errors.
fn prepare(device: &cubecl::Device, path: &Path) -> Result<Input, String> {
    let image = decode(path)?;
    let size = [image.width(), image.height()];
    let [width, height] = size;
    let frame = Frame::new(width, height, image.samples())?;
    let name = path.file_name().and_then(std::ffi::OsStr::to_str)
        .ok_or("source has no UTF-8 basename")?.to_owned();
    let proxy = thumbnail_with_scale(device, frame, width, height, 6, 268_435_456)?;
    Ok(Input { name, size, proxy })
}

/// Record both directions with each estimate's actual confidence and geometry.
/// # Errors
/// Returns a product score or transform error.
fn compare_both(
    device: &cubecl::Device,
    role: &str,
    before: &Input,
    after: &Input,
    estimates: &mut Vec<Value>,
) -> Result<(), String> {
    for (reference, source) in [(before, after), (after, before)] {
        let fit = estimate_pair(device, &reference.proxy, &source.proxy, Model::Breathing, 0.05, false)?;
        estimates.push(serde_json::json!({
            "role": role, "reference": reference.name, "source": source.name,
            "transform": fit.transform.matrix(), "confidence": fit.confidence,
        }));
    }
    Ok(())
}
