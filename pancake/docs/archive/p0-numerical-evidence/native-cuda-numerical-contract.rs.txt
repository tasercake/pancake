//! Native cuda execution of the unchanged independent numerical contracts.

#[path = "common/numerical_cases.rs"]
pub mod numerical_cases;

use slint as _;
use wgpu as _;

/// Execute exact variance, shape, logical budget and completed-job reuse checks.
///
/// # Errors
/// Returns device, runtime, numerical, ownership or artifact-write failures.
#[test]
fn native_cuda_variance_budget_and_reuse() -> Result<(), String> {
    let device = cubecl::Device::cuda(0).map_err(|error| error.to_string())?;
    numerical_cases::check_variance(&device)?;
    numerical_cases::check_budget_and_reuse(&device, "native-cuda")
}
