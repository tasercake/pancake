"""Match the canonical sanitizer test executions to exact CPU dev JUnit."""
import collections
import gzip
from hashlib import sha256
import json
from pathlib import Path
import re
import sys
import xml.etree.ElementTree as ET

root = Path(sys.argv[1])
expected_revision = "11cfcf8aac5921dfa843bfc5777eb4b2643a6ca6"
receipt = json.loads((root / "receipt.json").read_text())
assert receipt["source_revision"] == expected_revision and receipt["gate_exit_status"] == 0
assert receipt["passing"] and receipt["totals"]["passed"] == 164
assert all(receipt["totals"][name] == 0 for name in ("failed", "ignored", "measured", "filtered_out"))
assert not receipt["diagnostic_lines"]

junit = ET.fromstring(gzip.decompress((root / "nextest-junit.xml.gz").read_bytes()))
assert (junit.get("tests"), junit.get("failures"), junit.get("errors"), junit.get("skipped")) == ("164", "0", "0", "0")
expected = {(suite.attrib["name"], case.attrib["name"]) for suite in junit.findall("testsuite") for case in suite.findall("testcase")}
assert len(expected) == 164

header = re.compile(r"^\s*Running (.+) \(([^()]+)\)$")
passed = re.compile(r"^test (.+) \.\.\. ok$")
raw = (root / "sanitizer-address.log").read_text()
observed = {}
groups = set()
current = None
for line in raw.splitlines():
    if found := header.match(line):
        source, elf = found.groups()
        package = elf.split("/build/", 1)[1].split("/", 1)[0]
        if source == "unittests src/lib.rs":
            suite = package
        elif source == "unittests src/main.rs":
            suite = f"{package}::bin/{package}"
        else:
            assert source.startswith("tests/") and source.endswith(".rs"), source
            suite = f"{package}::{Path(source).stem}"
        assert suite not in groups and elf in receipt["tested_and_application_elfs"], (suite, elf)
        groups.add(suite)
        current = (suite, elf)
    elif result := passed.match(line):
        assert current is not None, line
        suite, elf = current
        case = (suite, result[1])
        assert case not in observed, case
        observed[case] = elf
    elif line.startswith("test result:"):
        current = None
assert set(observed) == expected, (set(observed) - expected, expected - set(observed))
assert len(groups) == 20
selected = (root / "selected-targets.txt").read_text()
assert selected == (root / "nextest-selected-targets.txt").read_text()
assert len(selected.splitlines()) == 17 and "cpu_gui_native" not in selected
matches = {
    "source_revision": expected_revision,
    "matched_nextest_dev_names": len(expected),
    "sanitizer_roster_names": len(observed),
    "matched_groups": len(groups),
    "counts": dict(sorted(collections.Counter(suite for suite, _ in observed).items())),
    "nextest_junit_gzip_sha256": sha256((root / "nextest-junit.xml.gz").read_bytes()).hexdigest(),
    "sanitizer_log_sha256": sha256((root / "sanitizer-address.log").read_bytes()).hexdigest(),
    "tests": [{"suite": suite, "name": name, "elf_path": elf, "status": "passed"} for (suite, name), elf in sorted(observed.items())],
}
(root / "test-roster.json").write_text(json.dumps(matches, indent=2) + "\n")
print(json.dumps({key: value for key, value in matches.items() if key != "tests"}, indent=2))
