#!/usr/bin/env bash
set -euo pipefail
repo=$(git rev-parse --show-toplevel)
proof="$repo/artifacts/t50-tiff-p2"
cd "$repo/pancake"
test -z "$(git status --porcelain)"
git rev-parse HEAD > "$proof/revision.txt"
export CARGO_BUILD_JOBS=2 CARGO_INCREMENTAL=0 CARGO_PROFILE_DEV_DEBUG=0
export RUSTC_WRAPPER=/tmp/pancake-tools/bin/pancake-rust-cache
printf '%s\n' 'CARGO_BUILD_JOBS=2 CARGO_INCREMENTAL=0 RUSTC_WRAPPER=/tmp/pancake-tools/bin/pancake-rust-cache cargo build --release --locked -p pancake-cli --bin pancake-cli' > "$proof/build-command.txt"
cargo build --release --locked -p pancake-cli --bin pancake-cli > "$proof/build.log" 2>&1
{
  sha256sum target/release/pancake-cli Cargo.lock
  readelf -n target/release/pancake-cli
  rustc -Vv
} > "$proof/build-identity.txt"
