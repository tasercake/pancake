#!/usr/bin/env bash
# Prepare and optionally execute one exact full-roster subpartition of Pancake's CPU mutation gate.
set -euo pipefail

mode=${1:?Supply prepare or run}
output=${2:?Supply a fresh evidence directory}
if [[ $mode != prepare && $mode != run ]]; then
    printf 'Expected prepare or run\n' >&2
    exit 2
fi
records=$(realpath -m -- "$output")
repo=$(git rev-parse --show-toplevel)
cd "$repo/pancake"
source_revision=287828aac07ffad2252785b73a52a59fc07b3409
cargo_lock=f523404bf033b0532dc46ce71aa515dc398f56c32e6aee1c3529cf90a27018dc
mutator_sha=f985f265ee3ea3e453aa98b04c52134953911f692f8ce8abf137f3873202a2d0
cpu_targets_sha=d1180f38b292c12c077049872a7c8ce68a7517a643fcaecd71b07acdc7878f9e
export PATH="/tmp/pancake-tools/bin:$PATH"
export CARGO_NET_OFFLINE=true CARGO_BUILD_JOBS=2 CARGO_INCREMENTAL=0 CARGO_PROFILE_DEV_DEBUG=0
export RUSTC_WRAPPER=/tmp/pancake-tools/bin/pancake-rust-cache
export CARGO_TARGET_DIR="$repo/pancake/target/mutation-287828a"
unset DISPLAY WAYLAND_DISPLAY DBUS_SESSION_BUS_ADDRESS AT_SPI_BUS_ADDRESS
unset PANCAKE_RUNTIME PANCAKE_TEST_GPU_RUNTIME CONFORMANCE_EVENTS_PATH

if [[ $(git rev-parse HEAD) != "$source_revision" || -n $(git status --porcelain=v1 --untracked-files=all) ]]; then
    printf 'The exact 287828a clean checkout is required\n' >&2
    exit 2
fi
if [[ $(cargo mutants --version) != 'cargo-mutants 27.1.0' || $(rustc --version) != 'rustc 1.100.0 (0ef1bc386 2026-11-09)' ]]; then
    printf 'The pinned Rust and cargo-mutants versions are required\n' >&2
    exit 2
fi
printf '%s  %s\n' "$cargo_lock" Cargo.lock "$mutator_sha" "$(command -v cargo-mutants)" | sha256sum -c
mkdir -- "$records"
{
    git rev-parse HEAD
    sha256sum Cargo.lock "$(command -v cargo-mutants)"
    rustc -Vv
    cargo -V
    cargo mutants --version
    uname -srmo
    lscpu | awk '/Model name/ && !seen++ {print}'
    printf 'CARGO_BUILD_JOBS=%s\nCARGO_INCREMENTAL=%s\nCARGO_PROFILE_DEV_DEBUG=%s\nCARGO_TARGET_DIR=%s\nRUSTC_WRAPPER=%s\n' \
        "$CARGO_BUILD_JOBS" "$CARGO_INCREMENTAL" "$CARGO_PROFILE_DEV_DEBUG" "$CARGO_TARGET_DIR" "$RUSTC_WRAPPER"
} >"$records/provenance.txt"
git status --porcelain=v1 --untracked-files=all >"$records/source-status-before.txt"
cargo metadata --locked --no-deps --format-version 1 >"$records/metadata.json"
python3 scripts/assurance_targets.py "$records/metadata.json" cpu >"$records/cpu-targets.txt"
printf '%s  %s\n' "$cpu_targets_sha" "$records/cpu-targets.txt" | sha256sum -c
selectors=(--lib --bins)
while IFS= read -r target; do selectors+=(--test "$target"); done <"$records/cpu-targets.txt"
mutation=(cargo mutants --workspace --test-workspace false --cargo-arg=--workspace
    --cap-lints true --all-features --in-place --timeout 60 --build-timeout 180
    --jobserver-tasks 2 --no-shuffle)
test_args=(--locked "${selectors[@]}" -- --include-ignored)
"${mutation[@]}" --list --json -- "${test_args[@]}" >"$records/full.json" 2>"$records/full.stderr"
"${mutation[@]}" --shard 15/16 --sharding round-robin --list --json \
    -- "${test_args[@]}" >"$records/parent.json" 2>"$records/parent.stderr"
"${mutation[@]}" --shard 255/256 --sharding round-robin --list --json \
    -- "${test_args[@]}" >"$records/list.json" 2>"$records/list.stderr"
python3 - "$records" >"$records/partition.json" <<'PY'
import hashlib
import json
import sys
from pathlib import Path

directory = Path(sys.argv[1])
expected = {
    "full.json": "a97e2a6c014d47a908afc1a786576f9b49230724896d1f6ca3e62af1b42ebc64",
    "parent.json": "09da8e48b30529f42100245fecf3615c483d7f79d944ef524a651c30f0b8f05b",
    "list.json": "b2b1c29114e199902fa9ec35dbf4d4cc2d71c1c410547234f8969c4e5af02bde",
}
def read_verified(name):
    raw = (directory / name).read_bytes()
    if hashlib.sha256(raw).hexdigest() != expected[name]:
        raise SystemExit(f"Unexpected pinned roster bytes for {name}")
    return json.loads(raw)

full = read_verified("full.json")
parent = read_verified("parent.json")
shard = read_verified("list.json")
if parent != full[15::16] or shard != full[255::256] or shard != parent[15::16]:
    raise SystemExit("Selected native partition does not match its exact roster and 16-grid parent")
if len(full) != 2583 or len(parent) != 161 or len(shard) != 10:
    raise SystemExit("Unexpected full, parent or native partition count")
print(json.dumps({"full_candidates": len(full), "parent_index": "15/16", "parent_candidates": len(parent),
    "selected_index": "255/256", "selected_candidates": len(shard),
    "selected_full_roster_ordinals": list(range(255, len(full), 256)), "listing_sha256": expected}, indent=2))
PY
if [[ $mode == prepare ]]; then
    git status --porcelain=v1 --untracked-files=all >"$records/source-status-after.txt"
    test ! -s "$records/source-status-after.txt"
    cat "$records/partition.json"
    exit 0
fi

invocation=("${mutation[@]}" --shard 255/256 --sharding round-robin --output "$records" -- "${test_args[@]}")
printf '%q ' "${invocation[@]}" >"$records/run.argv"
printf '\n' >>"$records/run.argv"
date --iso-8601=seconds >"$records/start.txt"
df -B1 --output=avail "$repo" >"$records/disk-before.txt"
set +e
/usr/bin/time -v -o "$records/run.time" "${invocation[@]}" 2>&1 | tee "$records/terminal.log"
status=("${PIPESTATUS[@]}")
set -e
printf '%s\n' "${status[0]}" >"$records/exit.status"
printf '%s\n' "${status[1]}" >"$records/log-writer.status"
date --iso-8601=seconds >"$records/end.txt"
df -B1 --output=avail "$repo" >"$records/disk-after.txt"
if [[ -d $CARGO_TARGET_DIR ]]; then
    du -sB1 -- "$CARGO_TARGET_DIR" >"$records/target-allocated-bytes.txt"
fi
git rev-parse HEAD >"$records/source-after.txt"
git status --porcelain=v1 --untracked-files=all >"$records/source-status-after.txt"
if [[ -s $records/mutants.out/outcomes.json ]]; then
    cp -- "$records/mutants.out/outcomes.json" "$records/outcomes.json"
fi

test "${status[1]}" = 0
test ! -s "$records/source-status-after.txt"
test "$(cat "$records/source-after.txt")" = "$source_revision"
if [[ ! -s $records/outcomes.json || ! -s $records/mutants.out/mutants.json ]]; then
    printf 'Native mutation artifacts are incomplete\n' >&2
    if [[ "${status[0]}" != 0 ]]; then exit "${status[0]}"; fi
    exit 1
fi
python3 - "$records" "$PWD/scripts" "${status[0]}" <<'PYTHON' >"$records/native-scope-outcomes.json"
from collections import Counter
import json
from pathlib import Path
import sys

directory, scripts = map(Path, sys.argv[1:3])
native_exit = int(sys.argv[3])
sys.path.insert(0, str(scripts))
from verify_mutation_shards import mutation_id, verify_shards

listing = directory / "list.json"
native = directory / "mutants.out/mutants.json"
expected_rows = json.loads(listing.read_bytes())
if json.loads(native.read_bytes()) != expected_rows:
    raise SystemExit("Native execution roster differs from independent exact subpartition")
targets = [item for line in (directory / "cpu-targets.txt").read_text().splitlines()
           for item in ("--test", line)]
tail = ["--locked", "--lib", "--bins", *targets, "--", "--include-ignored"]
report = json.loads((directory / "outcomes.json").read_bytes())
if report["cargo_mutants_version"] != "27.1.0":
    raise SystemExit("Native outcomes use a different cargo-mutants version")
expected = Counter(mutation_id(row) for row in expected_rows)
observed = Counter()
counts = Counter()
baseline = []
for row in report["outcomes"]:
    stages = []
    for stage in row["phase_results"]:
        argv = stage["argv"]
        if argv.count("--workspace") != 1 or "--all-features" not in argv:
            raise SystemExit("Native phase lacks the unchanged full-workspace scope")
        if stage["phase"] == "Test" and argv[-len(tail):] != tail:
            raise SystemExit("Native test phase differs from the exact CPU selectors")
        stages.append((stage["phase"], stage["process_status"]))
    if row["scenario"] == "Baseline":
        baseline.append({"summary": row["summary"], "stages": stages})
        if row["summary"] == "Success" and stages != [("Build", "Success"), ("Test", "Success")]:
            raise SystemExit("Successful baseline lacks both successful native phases")
    else:
        variant = row["scenario"]["Mutant"]
        observed[mutation_id(variant)] += 1
        counts[row["summary"]] += 1
if observed - expected:
    raise SystemExit("Native outcomes contain an unexpected or duplicate mutation")
print(json.dumps({"scope": "255/256 subpartition of the full roster", "native_exit": native_exit,
    "baseline": baseline, "selected_candidates": len(expected), "observed_candidates": sum(observed.values()),
    "missing_candidates": sum((expected-observed).values()), "native_outcomes": counts}, indent=2))
if native_exit == 0:
    verify_shards(listing, [directory])
PYTHON
exit "${status[0]}"
