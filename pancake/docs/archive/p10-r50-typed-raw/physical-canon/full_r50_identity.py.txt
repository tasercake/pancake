"""Bind the physical diagnostic to the completed CPU ELF pins."""

from protocol import sha256

PINS = {
    "stock": (
        "e81079dee4078d7a1149da0f5950aab8c89f8d44",
        "2d3ff9afaa17d73233fd2ef2ef635f999b53186b608f412dcf8ebd367d458c55",
    ),
    "candidate": (
        "c80d59abe0b47a5e831a1e2b4ccad132d96106c7",
        "f17bc8241f776fafed9f01be42835a65293480eabb221adef6c21ae23322452d",
    ),
}

def require(condition, issue):
    if not condition:
        raise ValueError(issue)


def identities(arm_paths):
    records = {}
    for arm, path in arm_paths.items():
        revision, pin = PINS[arm]
        require(path.is_file() and sha256(path) == pin, f"{arm} ELF identity did not match")
        records[arm] = {"source_revision": revision, "sha256": pin, "path": str(path)}
    return records
