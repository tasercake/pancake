"""Extend the owned positive Canon CR3 stage to the exact 50-frame R50 set."""

import argparse
from concurrent.futures import ThreadPoolExecutor
import hashlib
import json
from pathlib import Path
import shutil
from urllib.request import urlopen

CAP = 2_147_483_648
BASE_BYTES = 17_682_772
FULL_BYTES = 442_098_996
MANIFEST = "dd7a7e8a396d16e952d3d24d258afc1e05b77bb1664b179d3a43a7f66e413fab"
TARGETS = "58bb68533311c48696a3a844d80eef2f37a3c1f39649305948c838b7002663dc"


def digest(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def check(condition, issue):
    if not condition:
        raise ValueError(issue)


def verify(path, row):
    check(path.is_file() and not path.is_symlink()
          and path.stat().st_size == row["size_bytes"] and digest(path) == row["sha256"],
          f"unverified staged original: {path}")
    return {"path": row["path"], "bytes": row["size_bytes"], "sha256": row["sha256"]}


def acquire(root, row):
    path = root / row["path"]
    check(path.resolve().is_relative_to(root) and path.suffix == ".CR3",
          f"invalid frozen path: {path}")
    check(not path.exists(), f"input already exists: {path}")
    partial = path.with_suffix(".CR3.part")
    checksum = hashlib.sha256()
    remaining = row["size_bytes"]
    try:
        with urlopen(row["url"], timeout=45) as source, partial.open("xb") as output:
            while remaining:
                block = source.read(min(remaining, 1024**2))
                check(bool(block), f"short source: {row['path']}")
                output.write(block)
                checksum.update(block)
                remaining -= len(block)
            check(not source.read(1) and checksum.hexdigest() == row["sha256"],
                  f"wrong source bytes: {row['path']}")
        partial.chmod(0o400)
        partial.rename(path)
    finally:
        partial.unlink(missing_ok=True)
    return verify(path, row)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("repo", type=Path)
    parser.add_argument("scope", type=Path)
    args = parser.parse_args()
    root = args.scope.resolve(strict=True)
    inputs = (root / "inputs").resolve(strict=True)
    check(inputs.is_relative_to(root) and inputs.parent == root, "unowned inputs root")
    targets_path = args.repo / "pancake/docs/performance-targets.json"
    check(digest(targets_path) == TARGETS, "unexpected frozen targets identity")
    pin = json.loads(targets_path.read_bytes())["corpus_evidence"]["R50"]
    manifest = args.repo / pin["manifest_path"]
    check(digest(manifest) == pin["manifest_sha256"] == MANIFEST, "unexpected manifest identity")
    frozen = json.loads(manifest.read_bytes())
    rows = frozen["frames"]
    total = sum(row["size_bytes"] for row in rows)
    check(len(rows) == len({row["sha256"] for row in rows}) == 50 and
          total == frozen["selected_native_bytes"] == FULL_BYTES <= CAP ==
          frozen["native_download_budget_bytes"], "frozen corpus boundaries disagree")
    relative = [row["path"] for row in rows]
    check(relative == sorted(relative) and len(set(relative)) == 50, "bad frame order")
    for row in rows:
        destination = inputs / row["path"]
        check(destination.resolve().is_relative_to(inputs) and destination.suffix == ".CR3",
              f"invalid frozen input destination: {destination}")

    old = json.loads((root / "acquisition.json").read_bytes())
    check(old["manifest_sha256"] == MANIFEST, "two-frame receipt differs")
    carried = (rows[0], rows[-1])
    flat = [inputs / Path(row["path"]).name for row in carried]
    check(sorted(inputs.iterdir()) == sorted(flat), "two-frame stage contains unexpected paths")
    needed = total - sum(row["size_bytes"] for row in carried)
    check(needed + BASE_BYTES == FULL_BYTES and
          shutil.disk_usage(root).free >= 15 * (1 << 30) + needed,
          "not enough disk for the retained 15 GiB reserve")
    for row, local, receipt in zip(carried, flat, old["verified"], strict=True):
        check(receipt["name"] == local.name and receipt["sha256"] == row["sha256"] and
              receipt["bytes"] == row["size_bytes"], "substitute source in two-frame receipt")
        verify(local, row)
    for row, local in zip(carried, flat, strict=True):
        destination = inputs / row["path"]
        destination.parent.mkdir(parents=True, exist_ok=True)
        check(not destination.exists(), "destination unexpectedly exists")
        local.chmod(0o400)
        local.rename(destination)

    with ThreadPoolExecutor(max_workers=2) as pool:
        list(pool.map(lambda row: acquire(inputs, row), rows[1:-1]))
    ordered = [verify(inputs / row["path"], row) for row in rows]
    check({path.relative_to(inputs).as_posix() for path in inputs.rglob("*") if path.is_file()}
          == set(relative), "extra, duplicate or incomplete stage")
    receipt = {
        "scope": str(root), "inputs": str(inputs), "manifest_path": pin["manifest_path"],
        "manifest_sha256": MANIFEST, "performance_targets_sha256": TARGETS,
        "rights": frozen["source"], "ordered_inputs": ordered, "ordered_inputs_count": 50,
        "retained_without_duplicate": [row["path"] for row in carried],
        "retained_bytes": BASE_BYTES, "newly_downloaded_bytes": needed, "total_bytes": total,
        "minimum_free_space_reserve_bytes": 15 * (1 << 30),
    }
    with (root / "full-acquisition.json").open("x") as stream:
        json.dump(receipt, stream, indent=2)
        stream.write("\n")
    print(f"Verified {len(ordered)} originals, {total} bytes; {needed} newly downloaded")


if __name__ == "__main__":
    main()
