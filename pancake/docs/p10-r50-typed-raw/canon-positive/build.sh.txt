#!/usr/bin/env bash
set -euo pipefail
source_root=$(realpath "$1")
label=$2
scope=$(realpath "$3")
expected=$4
cd "$source_root/pancake"
test "$(git rev-parse HEAD)" = "$expected"
test -z "$(git status --porcelain --untracked-files=no)"
seal="$scope/build-$label"
mkdir "$seal"
export CARGO_TARGET_DIR="$scope/cargo-target"
export RUSTC_WRAPPER='' RUSTC_WORKSPACE_WRAPPER='' CARGO_INCREMENTAL=0
build=(/usr/bin/taskset -c 14,15 cargo +1.100.0 build --offline --locked --release --jobs 2 -p pancake -p pancake-cli --example canon_native_probe --bin pancake-cli)
printf '%q ' "${build[@]}" > "$seal/command"
printf '\n' >> "$seal/command"
{
    git rev-parse HEAD 'HEAD^{tree}'
    sha256sum Cargo.lock examples/canon_native_probe.rs
    rustc +1.100.0 -vV
    printf 'RUSTC_WRAPPER=%q RUSTC_WORKSPACE_WRAPPER=%q CARGO_INCREMENTAL=%q\n' "$RUSTC_WRAPPER" "$RUSTC_WORKSPACE_WRAPPER" "$CARGO_INCREMENTAL"
    uname -a
    date --iso-8601=seconds
} > "$seal/provenance.txt"
"${build[@]}" > "$seal/build.log" 2>&1
cp "$CARGO_TARGET_DIR/release/pancake-cli" "$CARGO_TARGET_DIR/release/examples/canon_native_probe" "$seal/"
(cd "$seal" && sha256sum pancake-cli canon_native_probe > sha256)
readelf -n "$seal/pancake-cli" "$seal/canon_native_probe" > "$seal/elf-notes.txt"
date --iso-8601=seconds >> "$seal/provenance.txt"
