//! Direct production-module probe for independent fixture and pixel verification.

use std::io::{self, Write};
use std::path::Path;

use pancake::image_io::decode;

/// Read the caller's true input and write dimensions, precision and f32 samples.
fn main() -> Result<(), Box<dyn std::error::Error>> {
    let mut arguments = std::env::args_os().skip(1);
    let path = arguments.next().ok_or_else(|| io::Error::other("missing path"))?;
    let frame = decode(Path::new(&path))?;
    let mut output = io::BufWriter::new(io::stdout().lock());
    writeln!(output, "{} {} {:?}", frame.width(), frame.height(), frame.sample_encoding())?;
    for sample in frame.samples() {
        output.write_all(&sample.to_le_bytes())?;
    }
    output.flush()?;
    Ok(())
}
