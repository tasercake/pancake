"""Bind one existing CLI call and a profiler path to the opened Linux ELF."""
import json
import os
from pathlib import Path
import re
import stat
import sys


def main():
    if sys.platform != "linux" or os.execve not in os.supports_fd:
        raise ValueError("Linux fd-based execve is required")
    config = json.loads(Path(sys.argv[1]).read_text())
    elf = config["elf"]
    program = Path(elf["path"])
    if not program.is_absolute() or program.resolve(strict=True) != program:
        raise ValueError("expected a canonical absolute ELF path")
    events = Path(os.environ["CONFORMANCE_EVENTS_PATH"])
    call_directory = events.parent.resolve(strict=True)
    call = call_directory.relative_to(Path(config["report_root"]).resolve(strict=True))
    if (events.name != "events.jsonl" or len(call.parts) != 3 or call.parts[0] != "cases"
            or not re.fullmatch(r"call-[0-9]+", call.name)):
        raise ValueError("expected an existing frozen CLI call directory")
    descriptor = os.open(program, os.O_RDONLY | os.O_NOFOLLOW | os.O_CLOEXEC)
    try:
        st = os.fstat(descriptor)
        if (not stat.S_ISREG(st.st_mode)
                or [st.st_dev, st.st_ino, st.st_size, st.st_mtime_ns] != elf["stat"]
                or os.pread(descriptor, 4, 0) != b"\x7fELF"):
            raise ValueError("the held ELF differs from preflight")
        raw = Path(config["raw_root"]).resolve(strict=True) / call
        raw.mkdir(parents=True)
        pid = os.getpid()
        argv = [str(program), *sys.argv[2:]]
        environment = dict(os.environ)
        environment["LLVM_PROFILE_FILE"] = str(raw / "pancake-%p-%b%c.profraw")
        record = {"pid": pid, "argv": argv, "call": call.as_posix(), "elf": elf,
                  "profile": str(raw / f"pancake-{pid}-{elf['build_id']}.profraw")}
        with (call_directory / "llvm-profile-launch.json").open("x") as output:
            json.dump(record, output, indent=2)
            output.write("\n")
        os.execve(descriptor, argv, environment)
    finally:
        os.close(descriptor)


if __name__ == "__main__":
    main()
