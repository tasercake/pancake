#!/usr/bin/env bash
set -uo pipefail
bundle=/tmp/pancake-assurance-25cc498
bash scripts/check.sh fast > "$bundle/fast.log" 2>&1
fast_rc=$?
printf '%s\n' "$fast_rc" > "$bundle/fast.status"
cargo metadata --locked --no-deps --format-version 1 > "$bundle/metadata.json" &&
    python3 scripts/assurance_targets.py "$bundle/metadata.json" cpu > "$bundle/selected-targets.txt" || exit "$?"
assurance_args=(--lib --bins)
while IFS= read -r target; do
    assurance_args+=(--test "$target")
done < "$bundle/selected-targets.txt"
cargo nextest run --locked --workspace "${assurance_args[@]}" --all-features \
    --cargo-profile dev --profile ci --no-tests=pass --run-ignored all \
    > "$bundle/cpu-dev.log" 2>&1
cpu_rc=$?
printf '%s\n' "$cpu_rc" > "$bundle/cpu-dev.status"
printf 'fast=%s cpu-dev=%s\n' "$fast_rc" "$cpu_rc"
if (( fast_rc || cpu_rc )); then exit 1; fi
