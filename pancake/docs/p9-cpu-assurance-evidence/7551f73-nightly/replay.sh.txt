#!/usr/bin/env bash
# Run the exact pinned CPU branch coverage gate and keep its original status.
set -euo pipefail
root=$(git rev-parse --show-toplevel)
proof="$root/artifacts/cpu-coverage-7551f73"
cd "$root/pancake"
test "$(git rev-parse HEAD)" = 7551f73dbe5132f7e9723a9ac4b0c9144504a27a
test -z "$(git status --porcelain --untracked-files=all)"
{
    date -u +%FT%TZ
    git rev-parse HEAD HEAD^{tree}
    printf 'dirty=false\n'
    sha256sum Cargo.lock scripts/check.sh scripts/compare_assurance_coverage.py scripts/assurance_targets.py scripts/tool-versions.sh
    uname -srm
    lscpu
    rustc +nightly-2026-11-15 --version --verbose
    cargo +nightly-2026-11-15 --version --verbose
    cargo llvm-cov --version
    df -k "$root"
    awk '/^Cpus_allowed_list:/{print}' /proc/self/status
    printf 'cargo_build_jobs=%s\ncargo_incremental=%s\ncargo_profile_dev_debug=%s\n' "$CARGO_BUILD_JOBS" "$CARGO_INCREMENTAL" "$CARGO_PROFILE_DEV_DEBUG"
    printf 'RUSTFLAGS=%s\nRUSTC_WRAPPER=%s\nLLVM_PROFILE_FILE=%s\nRUST_TEST_THREADS=%s\n' \
       "${RUSTFLAGS-unset}" "${RUSTC_WRAPPER-unset}" "${LLVM_PROFILE_FILE-unset}" "${RUST_TEST_THREADS-unset}"
    printf 'DISPLAY=%s\nWAYLAND_DISPLAY=%s\nDBUS_SESSION_BUS_ADDRESS=%s\nAT_SPI_BUS_ADDRESS=%s\n' \
       "${DISPLAY-unset}" "${WAYLAND_DISPLAY-unset}" "${DBUS_SESSION_BUS_ADDRESS-unset}" "${AT_SPI_BUS_ADDRESS-unset}"
} >"$proof/environment.txt"
python3 scripts/miri_host_policy.py "$PWD" validate >"$proof/reviewed-source.log" 2>&1
date -u +%FT%TZ >"$proof/start"
set +e
/usr/bin/time -v -o "$proof/time.txt" bash scripts/check.sh branches-cpu >"$proof/branches-cpu.log" 2>&1
status=$?
set -e
printf '%s\n' "$status" >"$proof/branches-cpu.status"
date -u +%FT%TZ >"$proof/end"
exit "$status"
