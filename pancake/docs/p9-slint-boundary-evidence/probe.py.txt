"""Observe the embedded UI loading behavior on an owned Xvfb display."""
import argparse
import hashlib
import json
import os
from pathlib import Path
import select
import signal
import subprocess
import tempfile
import time

parser = argparse.ArgumentParser(description=__doc__)
parser.add_argument('binary', type=Path)
parser.add_argument('xvfb', type=Path)
parser.add_argument('output', type=Path)
args = parser.parse_args()
binary, server, output = (path.resolve() for path in (args.binary, args.xvfb, args.output))
output.mkdir(parents=True)
with binary.open('rb') as stream:
    sha = hashlib.file_digest(stream, 'sha256').hexdigest()
source = 'export component P9AuditMarker inherits Rectangle {}\n'
receipt = {'app': str(binary), 'binary_sha256': sha, 'xvfb': str(server),
           'settings': {'SLINT_BACKEND': 'winit-software'}, 'shadow_source': source, 'runs': []}
scrubbed = ('DISPLAY', 'WAYLAND_DISPLAY', 'XAUTHORITY', 'DBUS_SESSION_BUS_ADDRESS', 'DBUS_STARTER_ADDRESS',
            'DBUS_STARTER_BUS_TYPE', 'SESSION_MANAGER', 'AT_SPI_BUS_ADDRESS')

for variant in ('empty-cwd', 'local-widget-shadow'):
    with tempfile.TemporaryDirectory(prefix='pancake-p9-slint-') as folder:
        cwd = Path(folder)
        if variant.endswith('shadow'):
            (cwd / 'std-widgets.slint').write_text(source, encoding='utf-8')
        runtime = cwd / 'runtime'
        runtime.mkdir(mode=0o700)
        env = {key: value for key, value in os.environ.items() if key not in scrubbed}
        env.update(SLINT_BACKEND='winit-software', XDG_RUNTIME_DIR=str(runtime), HOME=str(cwd),
                   XDG_CONFIG_HOME=str(cwd / 'config'), XDG_DATA_HOME=str(cwd / 'data'),
                   XDG_CACHE_HOME=str(cwd / 'cache'), PANCAKE_STATE_DIR=str(cwd / 'state'),
                   PANCAKE_COMPUTE_CACHE_DIR=str(cwd / 'cache'))
        read_fd, write_fd = os.pipe()
        xargs = [str(server), '-displayfd', str(write_fd), '-screen', '0', '1280x900x24', '-nolisten', 'tcp']
        with (output / (variant + '-xvfb.log')).open('xb') as log:
            with subprocess.Popen(xargs, pass_fds=(write_fd,), env=env, stdout=log, stderr=log) as xvfb:
                try:
                    os.close(write_fd)
                    if not select.select([read_fd], [], [], 10)[0]:
                        raise RuntimeError('private Xvfb timed out')
                    display = os.read(read_fd, 64).decode().strip()
                    if not display.isdigit():
                        raise RuntimeError('private Xvfb did not allocate a display')
                    env['DISPLAY'] = ':' + display
                    app_log = output / (variant + '.log')
                    trace = output / (variant + '.strace')
                    command = ['dbus-run-session', '--', 'strace', '-f', '-e', 'trace=openat', '-o', str(trace), str(binary)]
                    with app_log.open('xb') as log:
                        with subprocess.Popen(command, cwd=cwd, env=env, stdout=log, stderr=subprocess.STDOUT,
                                              start_new_session=True) as app:
                            observed = ''
                            try:
                                until = time.monotonic() + 20
                                while app.poll() is None and time.monotonic() < until:
                                    windows = subprocess.run(['xwininfo', '-root', '-tree'], env=env,
                                                             capture_output=True, text=True, check=True, timeout=2).stdout
                                    if '("pancake-gui" "pancake-gui")' in windows:
                                        observed = windows
                                        break
                                    time.sleep(.05)
                                result = {'variant': variant, 'command': command, 'display': env['DISPLAY'],
                                          'exit_when_observed': app.poll(), 'window_observed': bool(observed),
                                          'log': app_log.name}
                                if observed:
                                    (output / (variant + '-windows.txt')).write_text(observed)
                            finally:
                                if app.poll() is None:
                                    os.killpg(app.pid, signal.SIGTERM)
                                app.wait(timeout=10)
                    openings = [line for line in trace.read_text(errors='replace').splitlines()
                                if 'std-widgets.slint' in line]
                    result['std_widget_openat_calls'] = openings
                    receipt['runs'].append(result)
                finally:
                    os.close(read_fd)
                    xvfb.terminate()
                    xvfb.wait(timeout=10)

with binary.open('rb') as stream:
    assert hashlib.file_digest(stream, 'sha256').hexdigest() == sha
(output / 'receipt.json').write_text(json.dumps(receipt, indent=2) + '\n')
print(json.dumps(receipt, indent=2))
