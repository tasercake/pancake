"""Evaluate the period-eight expansion witness without images or dependencies."""

import json
import math

BINOMIAL = (1, 4, 6, 4, 1)
PERIOD = 8


def filtered(samples, center, divisor):
    return sum(
        tap * samples[(center + i - 2) % len(samples)]
        for i, tap in enumerate(BINOMIAL)
    ) / divisor


def reduce(samples):
    return [filtered(samples, 2 * i, 16) for i in range(len(samples) // 2)]


def bilinear(samples):
    result = []
    for i, sample in enumerate(samples):
        result.extend((sample, (sample + samples[(i + 1) % len(samples)]) / 2))
    return result


def gaussian(samples):
    inserted = [0.0] * (2 * len(samples))
    inserted[::2] = samples
    return [filtered(inserted, i, 8) for i in range(len(inserted))]


def project(samples, harmonic, shift):
    return 2 / PERIOD * sum(
        sample * math.cos(math.tau * harmonic * (i - shift) / PERIOD)
        for i, sample in enumerate(samples)
    )


def witness(shift):
    sharp, softer = 0.32, 0.10
    blend = (sharp**2 + softer**2) / (sharp + softer)
    fine = sharp + 0.14 * (blend - sharp)
    coarse = sharp + 0.60 * (blend - sharp)
    f = [math.cos(math.tau * (i - shift) / PERIOD) for i in range(PERIOD)]
    quarter = reduce(reduce(f))
    results = {"quarter_grid": quarter, "eighth_grid": reduce(quarter)}
    for expansion in (bilinear, gaussian):
        expanded = expansion(expansion(quarter))
        mixed = [fine * v + (coarse - fine) * low for v, low in zip(f, expanded)]
        first = project(mixed, 1, shift)
        third = project(mixed, 3, shift)
        results[expansion.__name__] = {
            "expanded": expanded,
            "first": first,
            "third": third,
            "third_u16_codes": third * 65535,
        }
    return results


if __name__ == "__main__":
    aligned = witness(0)
    shifted = witness(2)
    gamma = math.cos(math.pi / 8)**4 * math.cos(math.pi / 4)**4
    delta = -0.0240952380952381
    assert math.isclose(aligned["quarter_grid"][0], gamma, abs_tol=1e-14)
    assert math.isclose(aligned["bilinear"]["third"],
                        delta * gamma * (1 - math.sqrt(0.5)) / 2,
                        abs_tol=1e-14)
    assert math.isclose(aligned["gaussian"]["third"],
                        delta * gamma * math.cos(3 * math.pi / 8)**4 / 2,
                        abs_tol=1e-14)
    assert max(abs(row["third"]) for row in
               (shifted["bilinear"], shifted["gaussian"])) < 1e-14
    print(json.dumps({"aligned": aligned, "joint_shift_two_pixels": shifted}, indent=2))
