"""No-application tests using historical events and synthetic failures."""

import copy
import json
import runpy
import tarfile
import unittest
from pathlib import Path


HERE = Path(__file__).resolve().parent
triplet = runpy.run_path(str(HERE / "weighted-cpu-triplet.py.txt"))
validate = triplet["check_completion"]


class TripletPreparation(unittest.TestCase):
    def setUp(self):
        archive = HERE.parent / "t50-weighted-cpu-e81079d/raw-sanitized-pair.tar.gz"
        with tarfile.open(archive) as data:
            self.events = {
                role: [json.loads(line) for line in data.extractfile(f"{role}/events.jsonl")]
                for role in ("reference", "candidate")
            }
        start = next(row for row in self.events["reference"] if row["event"] == "stack_start")
        self.frames = [Path(name) for name in start["input_order"]]
        self.output = {"shape": [3998, 5998, 3], "dtype": "uint16"}

    def test_historical_role_specific_completion(self):
        validate("reference", self.events["reference"], self.frames, self.output)
        validate("stock", self.events["candidate"], self.frames, self.output)
        validate("resident", self.events["candidate"], self.frames, self.output)

    def test_rejects_reversed_requested_order_and_wrong_reference_crop(self):
        with self.assertRaisesRegex(ValueError, "order"):
            validate("reference", self.events["reference"], self.frames[::-1], self.output)
        events = copy.deepcopy(self.events["reference"])
        next(row for row in events if row["event"] == "crop_complete")["bounds"] = [0, 2, 1, 1]
        with self.assertRaisesRegex(ValueError, "crop"):
            validate("reference", events, self.frames, self.output)

    def test_rejects_missing_or_changed_pancake_sequential_map(self):
        for replacement in (None, [triplet["IDENTITY"]] * 49):
            with self.subTest(replacement=replacement):
                events = copy.deepcopy(self.events["candidate"])
                next(row for row in events if row["event"] == "stack_complete")[
                    "sequential_transforms"
                ] = replacement
                with self.assertRaisesRegex(ValueError, "sequential"):
                    validate("resident", events, self.frames, self.output)

    def test_rejects_fallback_and_wrong_map_path(self):
        events = copy.deepcopy(self.events["candidate"])
        complete = next(row for row in events if row["event"] == "stack_complete")
        complete["fallback_reason"] = "fallback"
        with self.assertRaisesRegex(ValueError, "backend"):
            validate("stock", events, self.frames, self.output)
        complete["fallback_reason"] = None
        complete["alignment_transforms"]["wrong"] = complete["alignment_transforms"].pop(str(self.frames[0]))
        with self.assertRaisesRegex(ValueError, "maps"):
            validate("stock", events, self.frames, self.output)

    def test_rss_above_frozen_gap_remains_pending(self):
        sample = {
            "rss_sampler_affinity": [15], "peak_rss_bytes": 10,
            "observed_peak_rss_bytes": 10, "maximum_rss_sample_gap_ns": 20_000_000,
        }
        rss = triplet["eligible_rss"]
        self.assertEqual(rss(sample), 10)
        self.assertIsNone(rss({**sample, "maximum_rss_sample_gap_ns": 20_000_001}))
        self.assertIsNone(rss({**sample, "peak_rss_bytes": None}))
        self.assertIsNone(rss({**sample, "rss_sampler_affinity": [14]}))

    def test_missing_lease_permission_is_rejected(self):
        check = triplet["check_lease"]
        valid = {
            "exclusive_host_granted": True,
            "stage6_read_granted": True,
            "mutation_terminal": True,
            "profile": triplet["PROFILE"],
            "materialization_sha256": triplet["MATERIALIZATION"],
            "order": list(triplet["ORDER"]),
        }
        check(valid)
        for key in ("exclusive_host_granted", "stage6_read_granted", "mutation_terminal"):
            with self.subTest(key=key), self.assertRaisesRegex(ValueError, "lease"):
                check({**valid, key: False})

    def test_launch_arguments_are_reused_by_all_three_roles(self):
        specs, frames, profile = triplet["static_settings"]()
        self.assertEqual(len(frames), 50)
        self.assertEqual(specs["stock"], specs["resident"])
        expected = [str(path) for path in frames]
        for role, argv in specs.items():
            with self.subTest(role=role):
                self.assertEqual(argv[argv.index("--input") + 1: argv.index("--input") + 51], expected)
                self.assertIn("--cpu", argv)
                self.assertNotIn("--gpu", argv)
        self.assertEqual(profile.key, "T50.weighted_average.cpu")


if __name__ == "__main__":
    unittest.main()
