"""Synthetic bounded-policy checks; uses no real app, source media, or device."""
from pathlib import Path
import runpy
import unittest

operator = runpy.run_path(str(Path(__file__).with_name("limited-gates.py.txt")))
reduce_gates = operator["reduce_gates"]
SCHEDULE = operator["SCHEDULE"]


class LimitedGateTests(unittest.TestCase):
    def rows(self):
        return [
            {"kind": kind, "number": number, "target": role,
             "elapsed_ns": 100 if role == "reference" else (100 if kind == "cold" else 55),
             "peak_rss_bytes": 1000 if role == "reference" else 550, "valid": True}
            for kind, number, role in SCHEDULE
        ]

    def test_exact_bounded_schedule_passes_inclusive_limits(self):
        result = reduce_gates(self.rows())
        self.assertEqual(result["status"], "pass")
        self.assertEqual(result["ratios"]["warm"]["ratio"], 0.55)
        self.assertEqual(result["ratios"]["rss"]["ratio"], 0.55)
        self.assertEqual(result["ratios"]["cold"]["ratio"], 1.0)
        self.assertFalse(result["frozen_full_credit"])

    def test_duplicate_or_missing_job_has_no_pass(self):
        rows = self.rows()
        self.assertEqual(reduce_gates(rows[:-1])["status"], "pending")
        self.assertEqual(reduce_gates([rows[0], *rows])["status"], "fail")

    def test_any_missing_peak_keeps_rss_pending_even_in_warmup(self):
        rows = self.rows()
        rows[0]["peak_rss_bytes"] = None
        result = reduce_gates(rows)
        self.assertEqual(result["status"], "pending")
        self.assertEqual(result["ratios"]["rss"]["status"], "pending")

    def test_numerical_miss_is_never_relabelled(self):
        rows = self.rows()
        for row in rows:
            if row["target"] == "candidate":
                row["peak_rss_bytes"] = 551
        self.assertEqual(reduce_gates(rows)["status"], "fail")


if __name__ == "__main__":
    unittest.main()
