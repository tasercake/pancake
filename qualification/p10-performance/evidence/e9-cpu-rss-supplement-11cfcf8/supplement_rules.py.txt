"""Eligibility and live-CPU guard for one reduced-release RSS supplement."""

ROSTER = (
    ("cold", 1, "reference"), ("cold", 1, "candidate"),
    ("cold", 2, "candidate"), ("cold", 2, "reference"),
    ("cold", 3, "reference"), ("cold", 3, "candidate"),
)
PERIOD_NS = 3_000_000
LIMIT_NS = 20_000_000
RELEASE_SUPPLEMENT_LIMIT = 0.55


def evaluate_rss(samples):
    """Require the entire registered roster before computing the aggregate ratio."""
    if tuple((row["kind"], row["number"], row["target"]) for row in samples) != ROSTER:
        raise ValueError("all six exact scheduled jobs are required in frozen roster order")
    ineligible = [
        {
            "kind": row["kind"], "number": row["number"], "target": row["target"],
            "maximum_rss_sample_gap_ns": row.get("maximum_rss_sample_gap_ns"),
        }
        for row in samples
        if (
            row["valid"] is not True or row["resource_errors"] or row["returncode"] != 0
            or row["timed_out"] or row["sample_interval_ns"] != PERIOD_NS
            or type(row.get("maximum_rss_sample_gap_ns")) is not int
            or not 0 <= row["maximum_rss_sample_gap_ns"] <= LIMIT_NS
            or type(row.get("peak_rss_bytes")) is not int or row["peak_rss_bytes"] <= 0
        )
    ]
    result = {
        "scope": "six-job reduced-release CPU resource supplement",
        "status": "pending", "full_frozen_credit": False,
        "limit": RELEASE_SUPPLEMENT_LIMIT, "ratio": None, "ineligible": ineligible,
    }
    if ineligible:
        return result
    maxima = {
        role: max(row["peak_rss_bytes"] for row in samples if row["target"] == role)
        for role in ("reference", "candidate")
    }
    ratio = maxima["candidate"] / maxima["reference"]
    return {
        **result, "peak_rss_bytes": maxima, "ratio": ratio,
        "status": "pass" if ratio <= RELEASE_SUPPLEMENT_LIMIT else "fail",
    }


def overlap_busy(before, after, cpus, seconds, ticks_per_second, affinity):
    """Use only live per-task deltas, and retain only anonymous aggregates."""
    overlap = []
    for identity, end in after.items():
        if identity not in before or end <= before[identity]:
            continue
        task, _ = identity
        if cpus & affinity(task):
            overlap.append((end - before[identity]) / (seconds * ticks_per_second))
    highest = max(overlap, default=0)
    total = sum(overlap)
    return {
        "largest_overlap_core_share": highest,
        "sum_overlap_core_share": total,
        "quiet": highest < 0.20 and total < 0.75,
    }
