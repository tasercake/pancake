"""Check the diagnostic report contract with synthetic RSS observations only."""

from copy import deepcopy
from pathlib import Path
import runpy
import unittest


gate = runpy.run_path(str(Path(__file__).with_name("calibration-envelope.py.txt")))
validate = gate["calibration"]
ROLES = gate["ROLES"]
PERIOD = 3_000_000


def fixture():
    report = {
        "profile": "T50.weighted_average.cpu",
        "collector_commit": gate["COLLECTOR"],
        "sample_interval_ns": PERIOD,
        "jobs": {},
    }
    for role, rss in zip(ROLES, (100, 60, 49), strict=True):
        report["jobs"][role] = {
            "launch": {"sample_interval_ns": PERIOD, "rss_sampler_cpu": 15},
            "sample": {
                "sample_interval_ns": PERIOD,
                "rss_sampler_affinity": [15],
                "valid": True,
                "resource_errors": [],
                "elapsed_ns": 23_000_000,
                "maximum_rss_sample_gap_ns": 20_000_000,
                "rss_sample_count": 2,
                "observed_peak_rss_bytes": rss,
                "peak_rss_bytes": rss,
            },
            "observations": [
                {"rss_elapsed_ns": 3_000_000, "rss_bytes": rss, "processes": [42]},
                {"rss_elapsed_ns": 23_000_000, "rss_bytes": 0, "processes": []},
            ],
        }
    return report


class CalibrationEnvelope(unittest.TestCase):
    def test_20ms_exact_boundary_can_describe_one_triplet(self):
        summary = validate(fixture())
        self.assertEqual(summary["candidate/reference_rss_ratio"], 0.49)
        self.assertFalse(summary["frozen_credit"])
        self.assertIn("diagnostic", summary["status"])

    def test_reference_miss_keeps_both_reference_ratios_pending(self):
        report = fixture()
        ref = report["jobs"]["reference"]
        ref["sample"].update(
            elapsed_ns=23_000_001, maximum_rss_sample_gap_ns=20_000_001,
            peak_rss_bytes=None,
        )
        ref["observations"][1]["rss_elapsed_ns"] += 1
        summary = validate(report)
        self.assertIsNone(summary["candidate/reference_rss_ratio"])
        self.assertIsNone(summary["stock/reference_rss_ratio"])
        self.assertEqual(summary["candidate/stock_rss_ratio"], 49 / 60)
        self.assertEqual(summary["rss"]["reference"]["observed_peak_rss_bytes"], 100)

    def test_period_mismatch_rejected_in_report_launch_and_sample(self):
        report = fixture()
        for part in ("root", "launch", "sample"):
            candidate = deepcopy(report)
            target = candidate if part == "root" else candidate["jobs"]["reference"][part]
            target["sample_interval_ns"] = 8_000_000
            with self.subTest(part=part), self.assertRaisesRegex(ValueError, "period"):
                validate(candidate)

    def test_reported_peak_cannot_fabricate_missing_edge(self):
        report = fixture()
        report["jobs"]["stock"]["sample"]["elapsed_ns"] = 45_000_000
        with self.assertRaisesRegex(ValueError, "raw RSS"):
            validate(report)

    def test_out_of_order_or_missing_rows_fail(self):
        report = fixture()
        events = report["jobs"]["candidate"]["observations"]
        events.reverse()
        with self.assertRaisesRegex(ValueError, "order"):
            validate(report)
        events.clear()
        with self.assertRaisesRegex(ValueError, "raw RSS"):
            validate(report)

    def test_worker_affinity_and_collector_revision_are_required(self):
        for selector in ("affinity", "collector"):
            report = fixture()
            if selector == "affinity":
                report["jobs"]["candidate"]["sample"]["rss_sampler_affinity"] = [14]
            else:
                report["collector_commit"] = "wrong"
            with self.subTest(selector=selector), self.assertRaisesRegex(ValueError, selector):
                validate(report)


if __name__ == "__main__":
    unittest.main()
