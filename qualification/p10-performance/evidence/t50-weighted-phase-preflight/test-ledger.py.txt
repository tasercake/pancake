"""Device-free tests for return matching and weighted trace validation."""

import runpy
import unittest
from pathlib import Path

probe = runpy.run_path(str(Path(__file__).with_name("weighted-probe.py.txt")))
Ledger = probe["Ledger"]
summarize = probe["summarize"]


class WeightedPhaseLedger(unittest.TestCase):
    def test_nested_tail_call_closes_same_return_address_once(self):
        ledger = Ledger()
        ledger.enter(1, "filter", 11, 8, 10)
        ledger.enter(1, "aggregate", 11, 8, 15)
        self.assertTrue(ledger.leave(1, 11, 8, 42))
        self.assertFalse(ledger.leave(1, 11, 8, 49))
        self.assertEqual([(row["label"], row["parent"], row["elapsed_ns"]) for row in ledger.rows], [
            ("aggregate", "filter", 27), ("filter", None, 32),
        ])

    def test_other_thread_and_wrong_stack_pointer_never_close(self):
        ledger = Ledger()
        ledger.enter(9, "decode", 200, 104, 8)
        self.assertFalse(ledger.leave(8, 200, 104, 9))
        self.assertFalse(ledger.leave(9, 200, 96, 10))
        self.assertTrue(ledger.leave(9, 200, 104, 11))

    def test_disjoint_top_level_wall_does_not_double_count_children(self):
        ledger = Ledger()
        ledger.enter(1, "add", 100, 88, 1)
        ledger.enter(1, "wait", 99, 80, 2)
        self.assertTrue(ledger.leave(1, 99, 80, 12))
        self.assertTrue(ledger.leave(1, 100, 88, 16))
        counts, timings = summarize(ledger.rows, {"add": 1, "wait": 1})
        self.assertEqual(counts, {"add": 1, "wait": 1})
        self.assertEqual(timings["top_level_ns"], 15)
        self.assertEqual(timings["by_label_ns"], {"add": 15, "wait": 10})
        with self.assertRaises(ValueError):
            summarize(ledger.rows, {"add": 2, "wait": 1})

    def test_reversed_time_is_rejected(self):
        ledger = Ledger()
        ledger.enter(1, "add", 10, 8, 20)
        with self.assertRaises(ValueError):
            ledger.leave(1, 10, 8, 19)


if __name__ == "__main__":
    unittest.main()
