"""Apply the existing independent octave estimator to the exact optical pair."""

from importlib.machinery import SourceFileLoader
from importlib.util import module_from_spec, spec_from_loader
import json
from pathlib import Path
import sys

import numpy as np
import tifffile


prior = Path("qualification/p10-performance/evidence/d150-band-focus")
loader = SourceFileLoader("prior_octaves", str(prior / "spectral_d150.py.txt"))
spec = spec_from_loader(loader.name, loader)
metric = module_from_spec(spec)
loader.exec_module(metric)
folder = Path(sys.argv[1]).resolve(strict=True)
comparison = json.loads((folder / "comparison.json").read_bytes())
historical = json.loads((prior / "spectral-diagnostics.json").read_bytes())
products = ("package-287828a", "coherence")
images = {label: tifffile.imread(folder / label / "out.tif") for label in products}
results = []
for idx, (row, earlier) in enumerate(zip(comparison["regions"], historical, strict=True)):
    x, y, width, height = row["shape_xywh"]
    if row["shape_xywh"] != earlier["first_canvas_xywh"]:
        raise ValueError("prior spectral witness coordinates changed")
    witness = metric.spectra(tifffile.imread(folder / f"witness-{idx}.tif"))
    spectra = {
        name: metric.spectra(image[y : y + height, x : x + width])
        for name, image in images.items()
    }
    radius = np.sqrt(
        np.fft.rfftfreq(width)[None, :] ** 2 + np.fft.fftfreq(height)[:, None] ** 2
    )
    edges = earlier["frequency_edges_cyc_per_px"]
    bands = []
    for (lo, hi), archived in zip(zip(edges, edges[1:]), earlier["bands"], strict=True):
        mask = (radius >= lo) & (radius < hi)
        source = witness[mask]
        source_energy = float(np.vdot(source, source).real)
        result_band = {"band_cyc_per_px": [lo, hi], "historical_pinned_reference": archived["pinned-reference"]}
        for label, samples in spectra.items():
            target = samples[mask]
            target_energy = float(np.vdot(target, target).real)
            signed = float(np.vdot(source, target).real)
            result_band[label] = {
                "signed_projection": signed / source_energy,
                "phase_correlation": signed / np.sqrt(target_energy * source_energy),
                "amplitude_ratio": np.sqrt(target_energy / source_energy),
            }
        if abs(result_band["package-287828a"]["signed_projection"] - archived["stock"]["signed_projection"]) > 1e-12:
            raise ValueError("independent packaged phase did not reproduce the stock control")
        bands.append(result_band)
    results.append({"first_canvas_xywh": row["shape_xywh"], "bands": bands})
(folder / "octaves.json").write_text(json.dumps(results, indent=2, allow_nan=False) + "\n")
for region in results:
    print(region["first_canvas_xywh"])
    for band in region["bands"]:
        print(band["band_cyc_per_px"], *(f"{band[name]['signed_projection']:.6f}" for name in products))
