"""Compare packaged and phase-aware Hybrid on the previously frozen optical pixels."""

import argparse
import json
import os
from pathlib import Path
import shutil
import sys
import tarfile
from importlib.machinery import SourceFileLoader
from importlib.util import module_from_spec, spec_from_loader

import cv2
import numpy as np
import tifffile


ROOT = Path.cwd()
PRIOR = ROOT / "qualification/p10-performance/evidence/d150-band-focus"
SOURCE_SHA = "b5b862bf0ffa3b9a630373082f606ad8a8912d8d71e642d1b667a7928f58133a"
PACKAGE_CLI = "874c8683623131b5619858a9cba61cb6219bc2eae6d6bdb30ed1bc5d8634ce8e"


def load_shared():
    """Reuse the exact prior fixed-map preparation and four-core runner."""
    name = str(PRIOR / "replay_focused_d150.py.txt")
    loader = SourceFileLoader("prior_d150", name)
    spec = spec_from_loader(loader.name, loader)
    shared = module_from_spec(spec)
    loader.exec_module(shared)
    return shared


def historical_registration():
    """Authenticate the approved held-out input archive and recover TIFF digests."""
    checksums = {
        filename: digest
        for digest, filename in (
            row.split() for row in (PRIOR / "SHA256SUMS").read_text().splitlines()
        )
    }
    archive_path = PRIOR / "raw-diagnostics.tar.gz"
    if shared.frozen.sha(archive_path) != checksums["raw-diagnostics.tar.gz"]:
        raise ValueError("original fixed-input archive changed")
    if shared.frozen.sha(PRIOR / "heldout-plan.json") != checksums["heldout-plan.json"]:
        raise ValueError("preregistered optical regions changed")
    with tarfile.open(archive_path) as archive:
        row = archive.extractfile("focused40/registered.json")
        if row is None:
            raise ValueError("historical fixed image hashes not in evidence")
        return json.load(row)


def prepare(args):
    """Prepare exact prior registered pixels with the immutable borrowed sources."""
    frozen = shared.frozen
    if frozen.sha(args.maps) != SOURCE_SHA:
        raise ValueError("independent registration archive does not match prior maps")
    originals = frozen.authenticated_inputs(args.originals.resolve(strict=True))
    map_by_name = json.loads(args.maps.read_bytes())["alignment_transforms"]
    maps = [map_by_name[path.name] for path in originals]
    plan = json.loads((PRIOR / "heldout-plan.json").read_bytes())
    heldout = [
        row for row in plan["selection"]
        if all(36 <= index < 76 for index in row["adjacent_indices"])
    ]
    if len(heldout) != 2:
        raise ValueError("prior fixed slice must contain both preregistered optical witnesses")
    frozen_frames = historical_registration()
    args.output.mkdir(exist_ok=False)
    if shutil.disk_usage(args.output).free < 12 * 1024**3:
        raise ValueError("insufficient diagnostic scratch headroom")
    cv2.setNumThreads(1)
    _, fixed, witnesses = frozen.prepare(
        args.output / "common", originals, maps, range(36, 76),
        [row["first_canvas_xywh"] for row in heldout],
    )
    if fixed != frozen_frames:
        raise ValueError("fixed D150 TIFF/map hashes differ from the archived comparison")
    if frozen.authenticated_inputs(args.originals.resolve(strict=True)) != originals:
        raise ValueError("borrowed optical originals changed")
    patches = []
    for idx, (plan_row, witness) in enumerate(zip(heldout, witnesses, strict=True)):
        file = args.output / f"witness-{idx}.tif"
        tifffile.imwrite(file, witness[4], photometric="rgb")
        patches.append({
            "shape_xywh": plan_row["first_canvas_xywh"], "phase_source": witness[2],
            "phase_source_index": witness[1], "source_laplace_p90": witness[0],
            "patch_sha256": frozen.sha(file),
        })
    receipt = {
        "manifest_sha256": frozen.MANIFEST_SHA, "map_sha256": SOURCE_SHA,
        "ordered_original_count": 150, "original_hashes_before_after_match": True,
        "selection": [36, 76], "registered_frames_reproduce_archived_bytes": True,
        "registered_frames": fixed, "witnesses": patches,
        "versions": {"opencv": cv2.__version__, "numpy": np.__version__, "tifffile": tifffile.__version__},
    }
    (args.output / "prepare.json").write_text(json.dumps(receipt, indent=2) + "\n")
    print("verified 40 exact earlier TIFFs and both optical witnesses", flush=True)


def compare(args):
    """Render the same two real fusers and report unsigned, signed and pixel distances."""
    frozen = shared.frozen
    if frozen.sha(args.package_cli) != PACKAGE_CLI:
        raise ValueError("wrong packaged CLI")
    candidate = args.working_cli.resolve(strict=True)
    if frozen.sha(candidate) != args.candidate_sha:
        raise ValueError("wrong isolated candidate binary")
    if set(os.sched_getaffinity(0)) != set(range(4)):
        raise ValueError("the diagnostic orchestrator must also use the reserved CPUs 0–3")
    receipt = json.loads((args.output / "prepare.json").read_bytes())
    images = sorted((args.output / "common").glob("source-*.tif"))
    expected = [row["common_tiff_sha256"] for row in receipt["registered_frames"]]
    if len(images) != 40 or [frozen.sha(path) for path in images] != expected:
        raise ValueError("prepared optical frames are no longer the verified pixels")
    by_model = {}
    for label, binary in (("package-287828a", args.package_cli), ("coherence", candidate)):
        print(label, "started", flush=True)
        lane = args.output / label
        image = shared.run([binary], ROOT / "pancake", images, lane)
        by_model[label] = image
        print(label, "completed", flush=True)
    if [frozen.sha(path) for path in images] != expected:
        raise ValueError("a renderer modified the registered inputs")
    previous = json.loads((PRIOR / "focused40-results.json").read_bytes())
    regions, panels = [], []
    for idx, row in enumerate(receipt["witnesses"]):
        witness_path = args.output / f"witness-{idx}.tif"
        if frozen.sha(witness_path) != row["patch_sha256"]:
            raise ValueError("independent optical witness differs")
        patch = tifffile.imread(witness_path)
        score, phase = frozen.features(patch)
        witness = (score, row["phase_source_index"], row["phase_source"], phase, patch)
        values = {}
        stack = [patch]
        for label, image in by_model.items():
            measured = frozen.cropped(image, row["shape_xywh"])
            stack.append(measured)
            values[label] = frozen.phase_comparison(measured, witness)
        regions.append({
            **row, **values,
            "historical_live_pinned_reference_same_pixels": previous["pinned-reference"]["regions"][idx],
            "stock_prototype_roi_mae_u16": float(np.abs(
                stack[1].astype(np.int32) - stack[2].astype(np.int32)
            ).mean()),
        })
        panels.append(np.hstack(stack))
    comparison = {
        "scope": "fixed-input optical CPU diagnostic, no timing or resource gate credit",
        "matching_historical_input_hashes": receipt["registered_frames_reproduce_archived_bytes"],
        "all_common_inputs_unchanged": True,
        "outputs": {
            label: {
                "encoded_sha256": frozen.sha(args.output / label / "out.tif"),
                "events_sha256": frozen.sha(args.output / label / "events.jsonl"),
                "native_shape_hwc": list(image.shape), "dtype": image.dtype.name,
            }
            for label, image in by_model.items()
        },
        "regions": regions,
    }
    path = args.output / "comparison.json"
    path.write_text(json.dumps(comparison, indent=2, allow_nan=False) + "\n")
    rgb = (np.vstack(panels) // 257).astype(np.uint8)
    if not cv2.imwrite(str(args.output / "phase-review.png"), cv2.cvtColor(rgb, cv2.COLOR_RGB2BGR)):
        raise ValueError("could not retain the native phase review panel")
    print(json.dumps(regions, indent=2), flush=True)


def arguments():
    """Separate the one-core preparation from the exclusive four-core replay."""
    parser = argparse.ArgumentParser(description=__doc__)
    commands = parser.add_subparsers(dest="step", required=True)
    stage = commands.add_parser("prepare")
    for key in ("originals", "maps", "output"):
        stage.add_argument(key, type=Path)
    replay = commands.add_parser("compare")
    for key in ("package_cli", "working_cli", "output"):
        replay.add_argument(key, type=Path)
    replay.add_argument("candidate_sha")
    return parser.parse_args()


if __name__ == "__main__":
    shared = load_shared()
    config = arguments()
    if config.step == "prepare":
        prepare(config)
    else:
        compare(config)
