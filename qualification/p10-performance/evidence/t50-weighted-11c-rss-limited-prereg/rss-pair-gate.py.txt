"""Independently check the exact 2 ms sampled process-tree RSS pair."""

MAX_GAP_NS = 20_000_000
PERIOD_NS = 2_000_000
LIMIT = 0.55


def audit(sample, rows):
    elapsed = sample["elapsed_ns"]
    if (sample["valid"] is not True or sample["rss_sampler_affinity"] != [15]
            or sample["sample_interval_ns"] != PERIOD_NS or sample["resource_errors"]
            or type(elapsed) is not int or elapsed <= 0):
        raise ValueError("sample does not match the authorized valid 2 ms CPU process-tree scope")
    timestamps, observed = [], []
    for row in rows:
        at, rss, pids = (row[key] for key in ("rss_elapsed_ns", "rss_bytes", "processes"))
        if (type(at) is not int or at < 0 or not isinstance(pids, list)
                or any(type(pid) is not int or pid <= 0 for pid in pids)
                or (rss is not None and (type(rss) is not int or rss < 0))):
            raise ValueError("malformed raw RSS acquisition")
        if at > elapsed:
            continue
        if timestamps and at <= timestamps[-1]:
            raise ValueError("unordered raw RSS acquisition")
        timestamps.append(at)
        observed.append(rss)
    if not timestamps:
        raise ValueError("no actual in-job RSS acquisitions")
    edges = [0, *timestamps, elapsed]
    gap = max(end - start for start, end in zip(edges, edges[1:]))
    peak = max((rss for rss in observed if rss is not None), default=None)
    actual = {
        "maximum_rss_sample_gap_ns": gap, "rss_sample_count": len(timestamps),
        "observed_peak_rss_bytes": peak,
        "peak_rss_bytes": peak if peak and gap <= MAX_GAP_NS and None not in observed else None,
    }
    if any(sample[key] != value for key, value in actual.items()):
        raise ValueError("producer did not match independently reduced RSS")
    return actual


def gate(reference, candidate):
    ref, cand = reference["peak_rss_bytes"], candidate["peak_rss_bytes"]
    ratio = cand / ref if cand is not None and ref is not None else None
    status = "pending" if ratio is None else "pass" if ratio <= LIMIT else "fail"
    return {"scope": "separate matched-pair RSS-only release evidence", "frozen_full_credit": False,
            "limit": LIMIT, "status": status, "candidate_to_reference": ratio,
            "reference": reference, "candidate": candidate}
