"""Summarize the paired profile without substituting RSS estimates for gates."""

import csv
import hashlib
import json
import sys
from pathlib import Path

root = Path(sys.argv[1]).resolve(strict=True)
destination = Path(sys.argv[2]).resolve(strict=True)
report_path = root / "paired/report.json"
report = json.loads(report_path.read_bytes())
if not report.get("finished_at_unix_ns"):
    raise ValueError("report must be terminal")
if report["profile"] != "R50.hybrid.cpu":
    raise ValueError("expected frozen R50 Hybrid CPU profile")

fields = [
    "kind", "number", "target", "valid", "returncode", "elapsed_ns",
    "maximum_rss_sample_gap_ns", "peak_rss_bytes", "observed_peak_rss_bytes",
    "kernel_process_peak_rss_bytes", "rss_sample_count",
    "max_sample_acquisition_ns", "rss_sampler_affinity",
    "rss_sampler_thread_cpu_ns", "rgb16_width", "rgb16_height",
    "pixel_sha256", "invalid_reason",
]
with (destination / "samples.csv").open("w", newline="") as stream:
    writer = csv.DictWriter(stream, fields, lineterminator="\n")
    writer.writeheader()
    for job in report["samples"] + report["diagnostics"]:
        output = job.get("output", {})
        shape = output.get("shape", [None] * 3)
        row = {field: job.get(field) for field in fields}
        row.update(
            rss_sampler_thread_cpu_ns=job.get("sampler_threads_cpu_ns", {}).get("rss"),
            rgb16_height=shape[0],
            rgb16_width=shape[1],
            pixel_sha256=output.get("pixel_sha256"),
        )
        writer.writerow(row)


def maximum(observations, field):
    return max((row[field] for row in observations if row.get(field) is not None),
               default=None)


def cadence(jobs):
    successful = [job for job in jobs if job["valid"]]
    return {
        "job_count": len(jobs),
        "completed_valid_job_count": len(successful),
        "maximum_rss_sample_gap_ns": maximum(successful, "maximum_rss_sample_gap_ns"),
        "job_count_exceeding_rss_20_ms_gap": sum(
            job["maximum_rss_sample_gap_ns"] > 20_000_000 for job in successful
        ),
        "job_count_with_unavailable_process_tree_peak": sum(
            job["peak_rss_bytes"] is None for job in successful
        ),
        "maximum_rss_sample_acquisition_ns": maximum(
            successful, "max_sample_acquisition_ns"
        ),
        "maximum_observed_process_tree_rss_bytes": maximum(
            successful, "observed_peak_rss_bytes"
        ),
        "maximum_kernel_child_rss_bytes": maximum(
            successful, "kernel_process_peak_rss_bytes"
        ),
        "rss_collector_affinity_masks": sorted(
            {tuple(job["rss_sampler_affinity"]) for job in successful}
        ),
        "rss_collector_cpu_ns_sum": sum(
            job["sampler_threads_cpu_ns"]["rss"] for job in successful
        ),
        "wall_ns_sum": sum(job["elapsed_ns"] for job in successful),
        "distinct_pixel_hashes": sorted(
            {job["output"]["pixel_sha256"] for job in successful}
        ),
        "output_shapes_hwc": sorted(
            {tuple(job["output"]["shape"]) for job in successful}
        ),
    }


identity = report.get("targets", {})
summary = {
    "profile": report["profile"],
    "verdict": report["status"],
    "invalid_reason": report.get("invalid_reason"),
    "raw_report_sha256": hashlib.sha256(report_path.read_bytes()).hexdigest(),
    "performance_targets_sha256": report["performance_targets_sha256"],
    "provenance": identity,
    "hardware": report.get("hardware"),
    "capture_geometry": report.get("reference_raw_source_geometry"),
    "inputs_unchanged": report.get("inputs_unchanged"),
    "corpus": report.get("corpus"),
    "settings": report["settings"],
    "primary_job_count": len(report["samples"]),
    "paired_pixel_comparison_count": len(report["pixel_comparisons"]),
    "diagnostic_job_count": len(report["diagnostics"]),
    "measurement_distributions_ns": report.get("elapsed_ns"),
    "frozen_gates": report.get("gates"),
    "primary": {
        name: cadence([job for job in report["samples"] if job["target"] == name])
        for name in ("reference", "candidate")
    },
    "diagnostic": {
        job["target"]: {
            "valid": job["valid"],
            "matching_warm_pixel_hashes": job.get("matching_warm_pixel_hashes"),
            "backend": job.get("completion"),
            "pixel_sha256": job.get("output", {}).get("pixel_sha256"),
        }
        for job in report["diagnostics"]
    },
}
# Completion has 50 full transforms per target in the exact raw diagnostic
# receipt. Omit these duplicates in the summary while preserving backend fields.
for target in summary["diagnostic"].values():
    if target["backend"]:
        target["backend"] = {
            key: value
            for key, value in target["backend"].items()
            if key not in ("alignment_transforms", "sequential_transforms", "input_order")
        }
(destination / "summary.json").write_text(json.dumps(summary, indent=2) + "\n")
