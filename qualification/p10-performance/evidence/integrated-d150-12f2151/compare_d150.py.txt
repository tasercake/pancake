"""Compare a retained paired D150 export in common first-frame coordinates."""

import json
import math
import sys
from pathlib import Path

import numpy as np
import tifffile
from PIL import Image

run = Path(sys.argv[1]).resolve(strict=True)
report = json.loads((run / "report.json").read_bytes())
if not report.get("inputs_unchanged"):
    raise ValueError("D150 input verification was incomplete")
source_h, source_w, channels = report["frames"][0]["shape"]
if channels != 3 or not report["settings"]["auto_crop"]:
    raise ValueError("expected the frozen auto-cropped RGB D150 profile")

names = ("reference", "candidate")
images = {}
rectangles = {}
notes = {}
for name in names:
    output = run / "warm-01" / name / "complete.tif"
    image = tifffile.imread(output)
    if image.ndim != 3 or image.shape[2] != 3 or image.dtype != np.uint16:
        raise ValueError(f"expected genuine HWC RGB16 output {output}")
    images[name] = image
    out_h, out_w, _ = image.shape
    if name == "reference":
        events = [
            json.loads(line)
            for line in (run / "diagnostic-01" / name / "events.jsonl")
            .read_text()
            .splitlines()
        ]
        completed = [
            e for e in events if e["event"] == "stack_complete" and e["completed"]
        ]
        crop = [e["bounds"] for e in events if e["event"] == "crop_complete"]
        if len(completed) != 1 or len(crop) != 1:
            raise ValueError("reference lacks completed stack or crop evidence")
        if completed[0]["output_shape"] != [source_h, source_w, channels]:
            raise ValueError("reference pre-crop canvas differs from the first input")
        y, bottom, x, right = crop[0] or [0, 0, 0, 0]
        notes[name] = (
            "Reference diagnostic crop event plus matching first warm output geometry"
        )
    else:
        x, remainder_x = divmod(source_w - out_w, 2)
        y, remainder_y = divmod(source_h - out_h, 2)
        right, bottom = x, y
        if remainder_x or remainder_y:
            raise ValueError(
                "candidate output geometry is not consistent with its symmetric crop"
            )
        notes[name] = (
            "Candidate original canvas and observed width and height; auto-crop uses symmetric borders"
        )
    if (
        min(y, bottom, x, right) < 0
        or y + out_h + bottom != source_h
        or x + out_w + right != source_w
    ):
        raise ValueError(f"reported {name} crop and decoded output disagree")
    rectangles[name] = (x, y, x + out_w, y + out_h)

common_rect = tuple(
    (max if index < 2 else min)(rectangles[name][index] for name in names)
    for index in range(4)
)
left, top, right, bottom = common_rect
if min(right - left, bottom - top) < 256:
    raise ValueError("published pair lacks a usable shared first-frame area")
common = {}
for name, image in images.items():
    x, y, *_ = rectangles[name]
    common[name] = image[top - y : bottom - y, left - x : right - x]
a, b = (common[name] for name in names)


def metrics(reference, candidate):
    samples, changed, abs_sum, squares, peak = 0, 0, 0, 0.0, 0
    signed = np.zeros(3, dtype=np.int64)
    for y in range(0, reference.shape[0], 64):
        delta = np.subtract(
            candidate[y : y + 64], reference[y : y + 64], dtype=np.int32
        )
        signed += delta.sum((0, 1), dtype=np.int64)
        absolute = np.abs(delta, out=delta)
        samples += absolute.size
        abs_sum += int(absolute.sum(dtype=np.int64))
        squares += float(np.square(absolute, dtype=np.float64).sum())
        peak = max(peak, int(absolute.max()))
        changed += np.count_nonzero(absolute)
    return {
        "samples": int(samples),
        "mae_u16": abs_sum / samples,
        "rms_u16": math.sqrt(squares / samples),
        "max_u16": peak,
        "changed_sample_fraction": float(changed / samples),
        "signed_channel_mean_u16": (signed / (samples / 3)).tolist(),
    }


def detail_score(values):
    luma = (
        0.2126 * values[..., 0] + 0.7152 * values[..., 1] + 0.0722 * values[..., 2]
    ) / 65535
    laplace = (
        luma[:-2, 1:-1]
        + luma[2:, 1:-1]
        + luma[1:-1, :-2]
        + luma[1:-1, 2:]
        - 4 * luma[1:-1, 1:-1]
    )
    return float(np.quantile(np.abs(laplace), 0.9))


tile = 192
ranking = sorted(
    (
        (detail_score(a[y : y + tile, x : x + tile]), x, y)
        for y in range(32, a.shape[0] - tile - 31, tile)
        for x in range(32, a.shape[1] - tile - 31, tile)
    ),
    reverse=True,
)
regions = []
panel = Image.new("RGB", (2 * tile, 4 * tile))
for index, (score, x, y) in enumerate(ranking[:4]):
    reference, candidate = a[y : y + tile, x : x + tile], b[y : y + tile, x : x + tile]
    candidate_score = detail_score(candidate)
    regions.append(
        {
            "source_xywh": [left + x, top + y, tile, tile],
            "rgb": metrics(reference, candidate),
            "reference_luma_laplacian_absolute_p90": score,
            "candidate_luma_laplacian_absolute_p90": candidate_score,
            "candidate_reference_ratio": candidate_score / score if score else None,
        }
    )
    for column, values in enumerate((reference, candidate)):
        panel.paste(
            Image.fromarray((values // 257).astype(np.uint8)),
            (tile * column, tile * index),
        )
panel.save(run / "focus-review.png")
full = Image.new("RGB", (1200, 450))
for column, image in enumerate(images[name] for name in names):
    slide = Image.fromarray((image // 257).astype(np.uint8))
    slide.thumbnail((600, 450))
    full.paste(slide, (600 * column, 0))
full.save(run / "whole-review.png")
result = {
    "scope": "First warm actual RGB16 pair. Descriptive quality metrics with no fitted transform, resampling or new threshold.",
    "published_first_frame_canvas_ltrb": rectangles,
    "geometry_provenance": notes,
    "same_output_geometry": rectangles["reference"] == rectangles["candidate"],
    "shared_first_frame_extent_ltrb": common_rect,
    "shared_extent_coverage_first_source": (right - left)
    * (bottom - top)
    / (source_w * source_h),
    "shared_extent_rgb": metrics(a, b),
    "focus_regions_selection": "Four distinct 192px tiles with the highest reference absolute luma Laplacian 90th percentile on a fixed non-overlapping grid; common source coordinates.",
    "focus_regions": regions,
}
(run / "focus-comparison.json").write_text(
    json.dumps(result, indent=2, allow_nan=False) + "\n"
)
