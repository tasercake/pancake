"""Test exact RSS and external lineage using an isolated synthetic session."""

import json
import os
import platform
import re
import select
import subprocess
import sys
import tempfile
import time
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[2]))
from resources import _session_memory

CHILD = """
import hashlib
import json
import os
import signal
import sys
from pathlib import Path

Path(sys.argv[1]).write_text(str(os.getpid()))
read_ready, write_ready = os.pipe()
worker = os.fork()
if worker == 0:
    resident = bytearray(2 * 1024 * 1024)
    for offset in range(0, len(resident), 4096):
        resident[offset] = 1
    os.write(write_ready, b'A')
    signal.pause()
    os._exit(0)
try:
    assert os.read(read_ready, 1) == b'A'
    brief = os.fork()
    if brief == 0:
        os._exit(23)
    assert os.waitstatus_to_exitcode(os.waitpid(brief, 0)[1]) == 23
    print(json.dumps({
        'job': os.getpid(), 'worker': worker, 'brief': brief,
        'session': os.getsid(0), 'pid_namespace': os.readlink('/proc/self/ns/pid'),
        'mnt_namespace': os.readlink('/proc/self/ns/mnt'),
        'user_namespace': os.readlink('/proc/self/ns/user'),
        'uid': os.getresuid(), 'gid': os.getresgid(), 'groups': os.getgroups(),
        'gid_map': Path('/proc/self/gid_map').read_text().split(),
        'cap_eff': next(line.split()[1] for line in Path('/proc/self/status').read_text().splitlines() if line.startswith('CapEff:')),
        'environment_hash': hashlib.sha256(json.dumps(dict(os.environ), sort_keys=True).encode()).hexdigest(),
        'marker': os.environ['PANCAKE_NAMESPACE_PROBE'],
    }), flush=True)
    sys.stdin.buffer.read(1)
finally:
    os.kill(worker, signal.SIGTERM)
    os.waitpid(worker, 0)
"""


def receive(stream):
    if not select.select([stream], [], [], 3)[0]:
        raise TimeoutError("three seconds elapsed waiting for a synthetic process")
    return json.loads(stream.readline())


def samples(session):
    result = []
    for _ in range(32):
        start = time.monotonic_ns()
        members, rss = _session_memory(session)
        result.append(
            {
                "pids": sorted(members),
                "rss_bytes": rss,
                "wall_ns": time.monotonic_ns() - start,
            }
        )
    return result


def controller(origin):
    with subprocess.Popen(
        [sys.executable, "-c", CHILD, str(origin)],
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        start_new_session=True,
    ) as job:
        try:
            case = receive(job.stdout)
            assert case["job"] == case["session"] == job.pid
            case["local_scan"] = samples(job.pid)
            case["proc_count"] = sum(
                p.name.isdecimal() for p in Path("/proc").iterdir()
            )
            print(json.dumps(case), flush=True)
            if sys.stdin.read(1) != "m":
                raise ValueError("observer lost the migration handshake")
            print(json.dumps(samples(job.pid)), flush=True)
            if sys.stdin.read(1) != "q":
                raise ValueError("observer lost the completion handshake")
        finally:
            job.stdin.close()
            job.wait(timeout=3)
        assert job.returncode == 0


def host_identity(pid):
    proc = Path("/proc") / str(pid)
    status = dict(
        line.split(":", 1) for line in (proc / "status").read_text().splitlines()
    )
    return {
        "host_pid": pid,
        "pids": list(map(int, status["NSpid"].split())),
        "host_parent": int(status["PPid"]),
        "host_session": os.getsid(pid),
        "host_supplementary_groups": list(map(int, status["Groups"].split())),
        "pid_namespace": os.readlink(proc / "ns/pid"),
        "cgroup": (proc / "cgroup").read_text().strip(),
    }


def run(scope, binary, setup, trace_dir):
    origin = scope / "origin"
    destination = scope / "destination"
    origin.mkdir()
    destination.mkdir()
    argv = [*setup, sys.executable, binary, "controller", str(origin / "cgroup.procs")]
    env = os.environ | {"PANCAKE_NAMESPACE_PROBE": "identical-synthetic-job"}
    prefix = trace_dir / "trace"
    traced = [
        "strace",
        "-ff",
        "--kill-on-exit",
        "--decode-pids=pidns",
        "-o",
        str(prefix),
        "-e",
        "trace=clone,clone3,fork,vfork,execve,exit,exit_group,setsid,wait4,waitid",
        "--",
        *argv,
    ]
    try:
        with subprocess.Popen(
            traced, stdin=subprocess.PIPE, stdout=subprocess.PIPE, env=env
        ) as proc:
            try:
                case = receive(proc.stdout)
                host_pids = list(
                    map(int, (origin / "cgroup.procs").read_text().split())
                )
                identities = {
                    row["pids"][-1]: row for row in map(host_identity, host_pids)
                }
                main, worker = identities[case["job"]], identities[case["worker"]]
                assert (
                    main["pid_namespace"]
                    == worker["pid_namespace"]
                    == case["pid_namespace"]
                )
                assert (
                    worker["host_parent"] == worker["host_session"] == main["host_pid"]
                )
                outer_before = samples(main["host_pid"])
                (destination / "cgroup.procs").write_text(str(worker["host_pid"]))
                moved = host_identity(worker["host_pid"])
                assert moved["cgroup"].endswith("/destination")
                assert moved["host_session"] == main["host_pid"]
                proc.stdin.write(b"m")
                proc.stdin.flush()
                local_after = receive(proc.stdout)
                outer_after = samples(main["host_pid"])
                for local, outer in (
                    (case["local_scan"], outer_before),
                    (local_after, outer_after),
                ):
                    assert all(row["pids"] == sorted(identities) for row in local)
                    assert all(row["pids"] == sorted(host_pids) for row in outer)
                    assert {row["rss_bytes"] for row in local + outer} == {
                        local[0]["rss_bytes"]
                    }
                case.update(
                    command=traced,
                    host_identities=identities,
                    host_scan_before=outer_before,
                    host_scan_after=outer_after,
                    local_scan_after=local_after,
                    moved_host_worker=moved,
                )
                proc.stdin.write(b"q")
                proc.stdin.flush()
            finally:
                proc.stdin.close()
                try:
                    proc.wait(timeout=4)
                except subprocess.TimeoutExpired:
                    proc.kill()
                    proc.wait()
            assert proc.returncode == 0
        case["host_ptrace"] = {
            path.name.rsplit(".", 1)[1]: path.read_text()
            for path in sorted(trace_dir.glob("trace.*"))
        }
        assert any(
            "+++ exited with 23 +++" in trace for trace in case["host_ptrace"].values()
        )
        assert re.search(
            r"WEXITSTATUS\(s\) == 23", case["host_ptrace"][str(main["host_pid"])]
        )
        return case
    finally:
        origin.rmdir()
        destination.rmdir()


def main():
    if len(sys.argv) == 3 and sys.argv[1] == "controller":
        controller(Path(sys.argv[2]))
        return
    private_scope = Path("/proc/self/cgroup").read_text().strip().split(":", 2)[2]
    if not Path(private_scope).name.startswith("codex-pancake-pid-roster-"):
        raise ValueError("use a new task-owned delegated user scope")
    scope = Path("/sys/fs/cgroup") / private_scope.lstrip("/")
    setup = [
        "unshare",
        "--user",
        "--map-current-user",
        "--pid",
        "--fork",
        "--mount",
        "--mount-proc",
    ]
    with tempfile.TemporaryDirectory(prefix="pancake-pid-roster-") as scratch:
        ordinary = Path(scratch) / "ordinary"
        ordinary.mkdir()
        isolated = Path(scratch) / "isolated"
        isolated.mkdir()
        script = str(Path(__file__).resolve())
        original = run(scope, script, [], ordinary)
        namespaced = run(scope, script, setup, isolated)
    assert original["environment_hash"] == namespaced["environment_hash"]
    assert original["uid"] == namespaced["uid"] and original["gid"] == namespaced["gid"]
    print(
        json.dumps(
            {
                "kernel": platform.platform(),
                "ordinary": original,
                "namespaced": namespaced,
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    main()
