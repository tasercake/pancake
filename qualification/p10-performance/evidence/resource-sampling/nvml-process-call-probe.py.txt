"""Diagnose E9 driver query latency with the original NVML sampler and builds."""

import argparse
import gzip
import json
import platform
import subprocess
import sys
import tarfile
import tempfile
import time
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[2]))
from protocol import Campaign, corpus, sha256, verify_frames, verify_output
from resources import NvidiaMemory, measure

ROOT = Path(__file__).resolve().parents[4]
TARGETS = ROOT / "pancake/docs/performance-targets.json"
EVIDENCE = Path(__file__).resolve().parents[1]
UUID = "GPU-c7900bb5-e779-c07b-a37d-4adff841c8ac"
PCI = "0000:01:00.0"
RELEASE = "12f2151c8a198b5209c8f4d506dbd17fd9aaa74f"
BUILDS = {
    "wgpu-vulkan": (
        "integrated-e9-12f2151/raw-paired-runs.tar.gz",
        "integrated-e9-runs/vulkan/diagnostic-01/candidate/sample.json",
        "32caffd65ed785b1883c39ec69cfd726f2a6f836e42c653a6385dacf80267055",
    ),
    "cuda": (
        "integrated-e9-cuda-12f2151/raw-paired-cuda.tar.gz",
        "integrated-e9-cuda-run/cuda/diagnostic-01/candidate/sample.json",
        "45cb96ed124a3c8145e3c7cbbf836a3334580d14d63fde59596b393225910ec4",
    ),
}
CACHE_KEYS = (
    "HOME",
    "XDG_CACHE_HOME",
    "XDG_CONFIG_HOME",
    "TMPDIR",
    "CUDA_CACHE_PATH",
    "__GL_SHADER_DISK_CACHE_PATH",
    "MESA_SHADER_CACHE_DIR",
    "NUMBA_CACHE_DIR",
    "PANCAKE_STATE_DIR",
    "PANCAKE_COMPUTE_CACHE_DIR",
)


class MeasuredNvml(NvidiaMemory):
    def __init__(self, uuid):
        self.samples = []
        super().__init__(uuid)

    def sample_session(self, session):
        row = {"session": session, "calls": []}
        self.samples.append(row)
        pids, total = super().sample_session(session)
        row.update(session_pids=sorted(pids), total_bytes=total)
        return pids, total

    def _query(self, getter, operation):
        started = time.monotonic_ns()
        start_cpu = time.thread_time_ns()
        rows = super()._query(getter, operation)
        cpu = time.thread_time_ns() - start_cpu
        ended = time.monotonic_ns()
        self.samples[-1]["calls"].append(
            {
                "function": f"nvmlDeviceGet{operation}RunningProcesses_v3",
                "start_monotonic_ns": started,
                "end_monotonic_ns": ended,
                "thread_cpu_ns": cpu,
                "result_count": len(rows),
            }
        )
        return rows


def prepare(runtime, binary, frames):
    archive_name, member, expected_hash = BUILDS[runtime]
    binary_hash = sha256(binary)
    if binary_hash != expected_hash:
        raise ValueError(f"unexpected {runtime} binary SHA-256 {binary_hash}")
    with tarfile.open(EVIDENCE / archive_name) as archive:
        prior = json.load(archive.extractfile(member))
    output_index = prior["argv"].index("--output")
    assert output_index == len(frames) + 2
    assert prior["argv"][1] == "--input"
    assert all(
        old.endswith(new.name)
        for old, new in zip(prior["argv"][2:output_index], frames)
    )
    return binary_hash, prior, prior["argv"][output_index + 2 :]


def read_trace(directory, name):
    return [json.loads(line) for line in (directory / name).read_text().splitlines()]


def run(directory, runtime, binary, frames, prepared):
    binary_hash, prior, options = prepared
    output = directory / "complete.tif"
    argv = [
        str(binary),
        "--input",
        *map(str, frames),
        "--output",
        str(output),
        *options,
    ]
    env = prior["environment"].copy()
    for key in CACHE_KEYS:
        path = directory / key.lower()
        path.mkdir()
        env[key] = str(path)
    env["CONFORMANCE_EVENTS_PATH"] = str(directory / "events.jsonl")
    collector = MeasuredNvml(UUID)
    try:
        result = measure(
            argv, cwd=directory, env=env, directory=directory, timeout=15, gpu=collector
        )
    finally:
        collector.close()
    result["calls"] = collector.samples
    result["gpu_trace"] = read_trace(directory, "gpu-resources.jsonl")
    result["rss_trace"] = read_trace(directory, "resources.jsonl")
    result["stdout"] = (directory / "stdout.txt").read_text()
    result["stderr"] = (directory / "stderr.txt").read_text()
    result["argv"] = argv
    result["env"] = env
    result["binary_sha256"] = binary_hash
    events = read_trace(directory, "events.jsonl")
    result["completion"] = completion = next(
        row for row in events if row["event"] == "stack_complete"
    )
    result["output"] = verify_output(output, [339, 512, 3])
    assert result["returncode"] == 0 and not result["resource_errors"]
    assert len(result["calls"]) == len(result["gpu_trace"])
    assert completion["runtime"] == runtime and completion["actual_backend"] == "gpu"
    assert completion["physical_gpu"] and completion["device_pci_address"] == PCI
    assert not completion["fallback_reason"] and result["observed_peak_gpu_bytes"] > 0
    assert result["output"]["pixel_sha256"] == prior["output"]["pixel_sha256"]
    return result


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--input-root", type=Path, required=True)
    parser.add_argument("--vulkan-cli", type=Path, required=True)
    parser.add_argument("--cuda-cli", type=Path, required=True)
    parser.add_argument("--receipt", type=Path, required=True)
    parser.add_argument("--prepare-only", action="store_true")
    args = parser.parse_args()
    campaign = Campaign.load(TARGETS)
    records, source = corpus(ROOT, campaign, "E9")
    frames = verify_frames(args.input_root, records, 9)
    binaries = (
        ("wgpu-vulkan", args.vulkan_cli.resolve()),
        ("cuda", args.cuda_cli.resolve()),
    )
    prepared = {
        runtime: prepare(runtime, binary, frames) for runtime, binary in binaries
    }
    if args.prepare_only:
        print(json.dumps({runtime: item[0] for runtime, item in prepared.items()}))
        return
    gpu = subprocess.check_output(
        [
            "nvidia-smi",
            "-i",
            UUID,
            "--query-gpu=uuid,name,driver_version,pci.bus_id",
            "--format=csv,noheader",
        ],
        text=True,
    ).strip()
    receipt = {
        "scope": "NVML API subcall timing diagnostic with explicit application events",
        "release": RELEASE,
        "platform": platform.platform(),
        "gpu": gpu,
        "performance_targets_sha256": sha256(TARGETS),
        "input_source": source,
        "inputs": records,
        "results": {},
    }
    with gzip.open(args.receipt, "xt") as stream:
        try:
            for runtime, binary in binaries:
                with tempfile.TemporaryDirectory(prefix=f"nvml-{runtime}-") as scratch:
                    receipt["results"][runtime] = run(
                        Path(scratch), runtime, binary, frames, prepared[runtime]
                    )
            verify_frames(args.input_root, records, 9)
            receipt["inputs_unchanged"] = True
        finally:
            json.dump(receipt, stream, indent=2)


if __name__ == "__main__":
    main()
