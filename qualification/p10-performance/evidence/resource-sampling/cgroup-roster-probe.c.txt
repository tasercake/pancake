#define _GNU_SOURCE

#include <errno.h>
#include <fcntl.h>
#include <linux/sched.h>
#include <signal.h>
#include <stdint.h>
#include <sys/syscall.h>
#include <unistd.h>

/* Prototype a direct spawn into an existing cgroup without changing affinity. */
int pancake_cgroup_spawn(const char *group, char *const argv[], int input, int output) {
    int directory = open(group, O_RDONLY | O_DIRECTORY | O_CLOEXEC);
    if (directory < 0) {
        return -errno;
    }
    struct clone_args args = {
        .flags = CLONE_INTO_CGROUP,
        .exit_signal = SIGCHLD,
        .cgroup = (uint64_t)directory,
    };
    long pid = syscall(SYS_clone3, &args, sizeof(args));
    int error = errno;
    close(directory);
    if (pid == 0) {
        if (setsid() < 0 || dup2(input, STDIN_FILENO) < 0 || dup2(output, STDOUT_FILENO) < 0) {
            _exit(125);
        }
        execv(argv[0], argv);
        _exit(126);
    }
    return pid < 0 ? -error : (int)pid;
}
