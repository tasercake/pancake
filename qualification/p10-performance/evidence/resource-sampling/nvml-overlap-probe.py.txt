"""Controlled acquisition and reducer witnesses with no NVML device access."""

import gzip
import hashlib
import json
import sys
import threading
from concurrent.futures import ThreadPoolExecutor
from contextlib import nullcontext
from pathlib import Path
from types import SimpleNamespace

sys.path.insert(0, str(Path(__file__).resolve().parents[2]))
from resources import _UNAVAILABLE_MEMORY, NvidiaMemory, _summarize_samples

MS = 1_000_000
MEMBERS = {11, 12, 13}


class SyntheticMemory(NvidiaMemory):
    """Use the real PID merge with controlled, independently owned lists."""

    def __init__(self, compute, graphics):
        self.lib = SimpleNamespace(
            nvmlDeviceGetComputeRunningProcesses_v3=None,
            nvmlDeviceGetGraphicsRunningProcesses_v3=None,
        )
        self.snapshots = {"Compute": compute, "Graphics": graphics}

    def _query(self, _getter, operation):
        return [
            SimpleNamespace(pid=pid, memory_bytes=used)
            for pid, used in self.snapshots[operation]()
        ]


def collect_threads(shared_lock):
    """Hold one compute call and test whether a second full call can return."""
    first_inside = threading.Event()
    second_entered = threading.Event()
    let_first_return = threading.Event()
    lock = threading.Lock() if shared_lock else nullcontext()

    def slow_compute():
        with lock:
            first_inside.set()
            if not let_first_return.wait(timeout=2):
                raise TimeoutError("the control did not release the first call")
            return [(11, 80), (99, 1000)]

    def fast_compute():
        second_entered.set()
        with lock:
            return [(11, 90)]

    first = SyntheticMemory(slow_compute, lambda: [(11, 100), (12, 25)])
    second = SyntheticMemory(fast_compute, lambda: [(11, 100), (13, 20)])
    with ThreadPoolExecutor(max_workers=2) as executor:
        slow = executor.submit(first.sample, MEMBERS)
        try:
            assert first_inside.wait(timeout=2)
            fast = executor.submit(second.sample, MEMBERS)
            assert second_entered.wait(timeout=2)
            before_release = (
                fast.done() if shared_lock else fast.result(timeout=2) == 120
            )
            assert before_release is not shared_lock and not slow.done()
        finally:
            let_first_return.set()
        aggregates = [slow.result(), fast.result()]
    assert aggregates == [125, 120]
    return {
        "driver_lock_shared": shared_lock,
        "second_complete_before_release": before_release,
        "complete_aggregates_in_dispatch_order": aggregates,
    }


def row(begin_ms, end_ms, value):
    return {
        "started_ns": begin_ms * MS,
        "gpu_elapsed_ns": end_ms * MS,
        "gpu_bytes": value,
    }


def reduced(rows, elapsed_ns):
    """Drive the unchanged production gap and peak calculation."""
    summary = _summarize_samples(rows, elapsed_ns, gpu=True, errors=[])
    return {key: value for key, value in summary.items() if "gpu" in key}


def clock_witnesses():
    old = row(10, 42, 200)
    fresh = [row(7, 8, 100), row(23, 24, 100)]
    naive = reduced([*fresh, old], 58 * MS)
    safe_coverage = reduced(fresh, 58 * MS)
    assert (
        naive["peak_gpu_bytes"] == 200 and naive["maximum_gpu_sample_gap_ns"] == 18 * MS
    )
    assert safe_coverage["peak_gpu_bytes"] is None
    assert safe_coverage["maximum_gpu_sample_gap_ns"] == 34 * MS

    independent_fresh = [*fresh, row(31, 32, 110), row(39, 40, 130), row(47, 48, 110)]
    bypass = reduced(independent_fresh, 58 * MS)
    assert (
        bypass["peak_gpu_bytes"] == 130
        and bypass["maximum_gpu_sample_gap_ns"] == 16 * MS
    )

    # Each table is correct at its own read time. Their union never existed.
    mosaic = SyntheticMemory(lambda: [(11, 100)], lambda: [(12, 100)]).sample(MEMBERS)
    assert mosaic == 200

    late = row(21, 38, 999)
    after_exit = reduced([fresh[0], late], 33 * MS)
    backdated = reduced([fresh[0], row(21, 21, 999)], 33 * MS)
    assert after_exit == {
        "maximum_gpu_sample_gap_ns": 25 * MS,
        "gpu_sample_count": 1,
        "observed_peak_gpu_bytes": 100,
        "peak_gpu_bytes": None,
    }
    assert backdated["peak_gpu_bytes"] == 999

    unavailable = SyntheticMemory(
        lambda: [(11, 100)], lambda: [(12, _UNAVAILABLE_MEMORY)]
    ).sample(MEMBERS)
    assert unavailable is None
    missing = reduced([fresh[0], row(15, 16, unavailable)], 20 * MS)
    assert (
        missing["peak_gpu_bytes"] is None
        and missing["maximum_gpu_sample_gap_ns"] == 8 * MS
    )

    return {
        "overtaken_old_pair": {
            "rows_in_completion_order": [*fresh, old],
            "elapsed_ns": 58 * MS,
            "completion_only": naive,
            "coverage_without_unproven_old_pair": safe_coverage,
        },
        "independent_fresh_pairs": {
            "rows": independent_fresh,
            "elapsed_ns": 58 * MS,
            "verified_fresh_coverage": bypass,
        },
        "cross_time_aggregate": {
            "compute_read_bytes": {11: 100},
            "graphics_read_bytes": {12: 100},
            "free_then_allocate_between_queries": True,
            "actual_maximum_simultaneously_resident_bytes": 100,
            "mixed_aggregate_bytes": mosaic,
        },
        "late_pair": {
            "rows": [fresh[0], late],
            "elapsed_ns": 33 * MS,
            "normal": after_exit,
            "incorrect_backdating_to_dispatch": backdated,
        },
        "attributed_unavailable_value": missing,
    }


def archived_lock_bound(path):
    with gzip.open(path, "rt") as source:
        physical = json.load(source)
    result = {"sha256": hashlib.sha256(path.read_bytes()).hexdigest(), "profiles": {}}
    for runtime, record in physical["results"].items():
        prior_completion = 0
        elapsed = record["elapsed_ns"]
        for row, calls in zip(record["gpu_trace"], record["calls"], strict=True):
            compute, graphics = calls["calls"]
            compute_wall = compute["end_monotonic_ns"] - compute["start_monotonic_ns"]
            if compute_wall > 20 * MS:
                expected = min(row["gpu_elapsed_ns"], elapsed) - prior_completion
                assert expected == record["maximum_gpu_sample_gap_ns"]
                # Favor the hypothetical overlap: assume zero time after a common stall.
                acquire_bound = row["started_ns"] + compute_wall
                minimum_gap = min(elapsed, acquire_bound) - prior_completion
                # The outer completion was at or after the graphics subcall's end.
                latest_compute_start = row["gpu_elapsed_ns"] - (
                    graphics["end_monotonic_ns"] - compute["start_monotonic_ns"]
                )
                assert row["started_ns"] <= latest_compute_start
                shared_block_inside_job = min(
                    compute_wall, elapsed - latest_compute_start
                )
                assert minimum_gap > 20 * MS and shared_block_inside_job > 20 * MS
                result["profiles"][runtime] = {
                    "prior_complete_ns": prior_completion,
                    "compute_started_no_earlier_ns": row["started_ns"],
                    "compute_started_no_later_ns": latest_compute_start,
                    "compute_wall_ns": compute_wall,
                    "split_graphics_wall_ns": (
                        graphics["end_monotonic_ns"] - graphics["start_monotonic_ns"]
                    ),
                    "archived_max_gap_ns": expected,
                    "observed_job_exit_ns": elapsed,
                    "conditional_shared_stall_inside_job_ns": shared_block_inside_job,
                    "conditional_minimum_gap_if_second_query_waits_on_same_stall_ns": minimum_gap,
                }
                break
            prior_completion = row["gpu_elapsed_ns"]
        else:
            raise ValueError(f"no archived stall for {runtime}")
    return result


if __name__ == "__main__":
    print(
        json.dumps(
            {
                "synthetic_two_worker_control": [
                    collect_threads(shared) for shared in (False, True)
                ],
                "synthetic_clock_witnesses": clock_witnesses(),
                "physical_archive_read_only": archived_lock_bound(Path(sys.argv[1])),
            },
            indent=2,
        )
    )
