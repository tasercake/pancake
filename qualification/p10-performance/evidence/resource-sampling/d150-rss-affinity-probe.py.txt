"""Exercise RSS sampler placement with a short single-CPU Linux scheduler probe."""

import json
import os
import platform
import sys
import tempfile
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[2]))
from resources import measure


CHILD = """
import json
import os
import time

print(json.dumps(sorted(os.sched_getaffinity(0))), flush=True)
deadline = time.monotonic() + 0.34
pids = []
for _ in range(3):
    pid = os.fork()
    if pid == 0:
        break
    pids.append(pid)
while time.monotonic() < deadline:
    pass
if pid:
    for child in pids:
        os.waitpid(child, 0)
"""


def main():
    cpu = int(sys.argv[1])
    job_mask = sorted(os.sched_getaffinity(0))
    if len(job_mask) != 1 or cpu in job_mask:
        raise ValueError("use a single-CPU taskset mask and a separate RSS sampler CPU")
    jobs = []
    with tempfile.TemporaryDirectory(prefix="d150-rss-affinity-") as scratch:
        for index, sampler in enumerate((None, cpu, cpu, None)):
            directory = Path(scratch) / str(index)
            directory.mkdir()
            result = measure(
                [sys.executable, "-c", CHILD],
                cwd=directory,
                env=os.environ.copy(),
                directory=directory,
                timeout=1,
                gpu=None,
                rss_sampler_cpu=sampler,
            )
            mask = json.loads((directory / "stdout.txt").read_text())
            assert mask == job_mask == sorted(os.sched_getaffinity(0))
            assert result["returncode"] == 0 and not result["resource_errors"]
            jobs.append(
                {
                    "requested_cpu": sampler,
                    "child_affinity": mask,
                    "result": result,
                    "trace": [
                        json.loads(line)
                        for line in (directory / "resources.jsonl").read_text().splitlines()
                    ],
                }
            )
    print(json.dumps({"platform": platform.platform(), "jobs": jobs}, indent=2))


if __name__ == "__main__":
    main()
