"""Compare native and Python procfs RSS on the same stable process session."""

import ctypes as c
import json
import os
import platform
import signal
import subprocess
import sys
import time
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parents[2]))
from resources import _session_memory

CHILD = """
import json
import os
import signal

read_ready, write_ready = os.pipe()
pids = [os.getpid()]
for _ in range(3):
    pid = os.fork()
    if pid == 0:
        resident = bytearray(8 * 1024 * 1024)
        for offset in range(0, len(resident), 4096):
            resident[offset] = 1
        os.write(write_ready, b'x')
        signal.pause()
        os._exit(0)
    pids.append(pid)
ready = 0
while ready < 3:
    ready += len(os.read(read_ready, 3 - ready))
print(json.dumps(pids), flush=True)
for pid in pids[1:]:
    os.waitpid(pid, 0)
"""


def native_snapshot(library):
    function = c.CDLL(str(library.resolve())).pancake_session_rss
    function.argtypes = [
        c.c_int,
        c.POINTER(c.c_int),
        c.c_size_t,
        c.POINTER(c.c_size_t),
        c.POINTER(c.c_uint64),
    ]
    function.restype = c.c_int
    pids = (c.c_int * 4096)()

    def snapshot(session):
        count = c.c_size_t()
        rss = c.c_uint64()
        status = function(session, pids, len(pids), c.byref(count), c.byref(rss))
        if status:
            raise OSError(status, os.strerror(status))
        return set(pids[: count.value]), rss.value

    return snapshot


def main():
    native = native_snapshot(Path(sys.argv[1]))
    observations = []
    with subprocess.Popen(
        [sys.executable, "-c", CHILD], stdout=subprocess.PIPE, start_new_session=True
    ) as child:
        try:
            expected_pids = set(json.loads(child.stdout.readline()))
            for index in range(64):
                order = [("python", _session_memory), ("native", native)]
                if index % 2:
                    order.reverse()
                record = {}
                for name, snapshot in order:
                    start_wall = time.monotonic_ns()
                    start_cpu = time.thread_time_ns()
                    pids, rss = snapshot(child.pid)
                    cpu = time.thread_time_ns() - start_cpu
                    wall = time.monotonic_ns() - start_wall
                    assert pids == expected_pids, (name, pids, expected_pids)
                    record[name] = {"rss_bytes": rss, "wall_ns": wall, "cpu_ns": cpu}
                assert (
                    record["native"]["rss_bytes"] == record["python"]["rss_bytes"] > 0
                )
                observations.append(record)
        finally:
            for pid in expected_pids - {child.pid}:
                os.kill(pid, signal.SIGTERM)
            child.wait(timeout=3)
    print(
        json.dumps(
            {
                "platform": platform.platform(),
                "python": sys.version,
                "affinity": sorted(os.sched_getaffinity(0)),
                "session_pids": sorted(expected_pids),
                "visible_proc_pids": sum(
                    path.isdecimal() for path in os.listdir("/proc")
                ),
                "observations": observations,
            },
            indent=2,
        )
    )


if __name__ == "__main__":
    main()
