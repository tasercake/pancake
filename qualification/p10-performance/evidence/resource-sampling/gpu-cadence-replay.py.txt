"""Reconstruct strict NVIDIA completion gaps from original paired receipts."""

import collections
import gzip
import hashlib
import json
import sys
import tarfile
from pathlib import Path

LIMIT_NS = 20_000_000
T50_SAMPLE_PREFIX = "diagnostic/diagnostic-01/"


def analyze(record, trace):
    elapsed = record["elapsed_ns"]
    on_time = [row for row in trace if row["gpu_elapsed_ns"] <= elapsed]
    previous = 0
    gaps = []
    for row in on_time:
        at = row["gpu_elapsed_ns"]
        gaps.append((at - previous, "interior" if previous else "launch"))
        previous = at
    gaps.append((elapsed - previous, "exit"))
    longest = max(gaps)[0]
    assert longest == record["maximum_gpu_sample_gap_ns"]
    assert len(on_time) == record["gpu_sample_count"]
    first_late = trace[len(on_time)] if len(on_time) < len(trace) else None
    max_acquisition = max(
        (row["gpu_elapsed_ns"] - row["started_ns"] for row in on_time), default=0
    )
    gpu_pids = {pid for row in on_time for pid in row["processes"]}
    return {
        "longest_gap_ns": longest,
        "gap_counts": collections.Counter(
            location for gap, location in gaps if gap > LIMIT_NS
        ),
        "max_within_job_acquisition_ns": max_acquisition,
        "in_flight_at_exit": bool(first_late and first_late["started_ns"] <= elapsed),
        "last_completion_before_exit_ns": elapsed - previous,
        "first_post_exit_acquisition_ns": (
            first_late["gpu_elapsed_ns"] - first_late["started_ns"]
            if first_late
            else None
        ),
        "distinct_observed_session_gpu_pids": len(gpu_pids),
    }


def replay(path):
    runs = collections.defaultdict(dict)
    with tarfile.open(path) as archive:
        for member in archive:
            directory, _, name = member.name.rpartition("/")
            if name == "sample.json":
                runs[directory]["sample"] = json.load(archive.extractfile(member))
            elif name == "gpu-resources.jsonl.gz":
                with gzip.GzipFile(fileobj=archive.extractfile(member)) as rows:
                    runs[directory]["trace"] = [json.loads(line) for line in rows]
    result = {
        "archive_name": path.name,
        "archive_sha256": hashlib.sha256(path.read_bytes()).hexdigest(),
        "targets": {},
    }
    targets = collections.defaultdict(list)
    for directory, run in sorted(runs.items()):
        if "trace" not in run:
            continue
        if "/diagnostic-" in directory and not directory.startswith(T50_SAMPLE_PREFIX):
            continue
        row = {"directory": directory, **analyze(run["sample"], run["trace"])}
        targets[run["sample"]["target"]].append(row)
    for target, samples in sorted(targets.items()):
        counts = collections.Counter()
        for sample in samples:
            counts.update(sample["gap_counts"])
        result["targets"][target] = {
            "total_jobs": len(samples),
            "gaps_over_20_ms": counts,
            "jobs_with_any_gap_over_20_ms": sum(
                bool(row["gap_counts"]) for row in samples
            ),
            "failed_exit_gaps_with_call_in_flight": sum(
                row["gap_counts"]["exit"] and row["in_flight_at_exit"]
                for row in samples
            ),
            "worst_gap_ns": max(row["longest_gap_ns"] for row in samples),
            "samples": samples,
        }
    return result


if __name__ == "__main__":
    print(json.dumps([replay(Path(name)) for name in sys.argv[1:]], indent=2))
