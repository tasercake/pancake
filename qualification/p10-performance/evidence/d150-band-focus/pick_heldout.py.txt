"""Select D150 regions by correlated detail in adjacent source pairs only."""
from pathlib import Path
from importlib.machinery import SourceFileLoader
from importlib.util import module_from_spec, spec_from_loader
import json
import sys

import cv2
import numpy as np

root = Path.cwd()
source = root / "qualification/p10-performance/evidence/d150-hybrid-detail-prototype/run_fixed_d150.py.txt"
loader = SourceFileLoader("d150_fixed", str(source))
spec = spec_from_loader(loader.name, loader)
frozen = module_from_spec(spec)
loader.exec_module(frozen)
# This grid is fixed before reading any source or target pixels.
# Each 96px region is disjoint from the original reported four 192px patches.
corners = [(x, y) for y in (345, 675, 1045) for x in (670, 880, 1505, 1735)]
coords = [(x, y, 96, 96) for x, y in corners]
quadrants = [(x >= 1152, y >= 864) for x, y in corners]


def registered_roi(original, matrix, region):
    """Apply the frozen first-canvas map on only the independent tile."""
    x, y, width, height = region
    affine = np.asarray(matrix if matrix is not None else frozen.IDENTITY, np.float64).copy()
    affine[:, 2] += affine[:, :2] @ [x, y]
    bgr = cv2.warpAffine(
        original, affine, (width, height), flags=cv2.INTER_LANCZOS4 | cv2.WARP_INVERSE_MAP,
        borderMode=cv2.BORDER_REPLICATE,
    )
    return np.rint(np.clip(bgr[..., ::-1], 0, 255) * 257).astype(np.uint16)


def select(inputs, maps):
    """Pick coherent raw-source detail before running either fusion."""
    best = {}
    previous = None
    for index, (source, matrix) in enumerate(zip(inputs, maps, strict=True)):
        bgr = cv2.imread(str(source), cv2.IMREAD_UNCHANGED)
        if bgr is None or bgr.shape != (1728, 2304, 3) or bgr.dtype != np.uint8:
            raise ValueError(f"not native D150 RGB8 {source}")
        bgr = bgr.astype(np.float32)
        current = [frozen.features(registered_roi(bgr, matrix, region)) for region in coords]
        if previous is not None:
            for region, quadrant, (a, first), (b, second) in zip(
                coords, quadrants, previous, current, strict=True,
            ):
                denom = np.linalg.norm(first) * np.linalg.norm(second)
                correlation = float(np.dot(first, second) / denom) if denom > 0 else 0.0
                # Both images must support the same intensity transitions.
                # A single local impulse or high-frequency phase shift cannot alone win.
                score = min(a, b) * max(correlation, 0)
                if quadrant not in best or score > best[quadrant]["support"]:
                    best[quadrant] = {
                        "first_canvas_xywh": region, "adjacent_indices": [index - 1, index],
                        "adjacent_files": [inputs[index - 1].name, source.name],
                        "detail_p90": [a, b], "signed_phase_correlation": correlation,
                        "support": score, "witness_index": index - 1 if a >= b else index,
                    }
        previous = current
    return sorted(best.values(), key=lambda row: row["first_canvas_xywh"])


def main():
    """Record raw-only focus support and authenticate the borrowed originals."""
    inputs = frozen.authenticated_inputs(Path(sys.argv[1]).resolve(strict=True))
    matrices = frozen.pinned_maps(inputs)
    cv2.setNumThreads(1)
    chosen = select(inputs, matrices)
    if frozen.authenticated_inputs(Path(sys.argv[1]).resolve(strict=True)) != inputs:
        raise ValueError("borrowed sources changed")
    output = {
        "scope": "raw input and archived first-frame maps only, no test render, no reference output",
        "grid": coords, "source_count": len(inputs), "selection": chosen,
    }
    Path(sys.argv[2]).write_text(json.dumps(output, indent=2) + "\n")
    print(json.dumps(chosen, indent=2))


if __name__ == "__main__":
    main()
