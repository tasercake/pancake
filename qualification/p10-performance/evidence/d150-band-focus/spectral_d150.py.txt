"""Check witness phase by frequency in the already captured physical images."""
from pathlib import Path
import json
import sys
import numpy as np
import tifffile


def spectra(rgb):
    """Native luma spectrum with fixed edge taper and no fitted transform."""
    luma = rgb @ np.asarray([0.2126, 0.7152, 0.0722]) / 65535
    height, width = luma.shape
    return np.fft.rfft2((luma - luma.mean()) * np.outer(np.hanning(height), np.hanning(width)))


def main():
    folder = Path(sys.argv[1])
    models = ("stock", "rank-only", "band-local", "pinned-reference")
    baseline = json.loads((folder / "stock-result.json").read_bytes())
    height = baseline["regions"][0]["first_canvas_xywh"][3]
    width = baseline["regions"][0]["first_canvas_xywh"][2]
    fx = np.fft.rfftfreq(width)[None, :]
    fy = np.fft.fftfreq(height)[:, None]
    radius = np.sqrt(fx * fx + fy * fy)
    octave_edges = (1/32, 1/16, 1/8, 1/4, 1/2)
    bands = [(radius >= low) & (radius < high) for low, high in zip(octave_edges, octave_edges[1:])]
    outputs = {name: tifffile.imread(folder / name / "out.tif") for name in models}
    result = []
    for index, row in enumerate(baseline["regions"]):
        x, y, width, height = row["first_canvas_xywh"]
        phase_truth = spectra(tifffile.imread(folder / f"witness-{index}.tif"))
        patch = {name: spectra(pixels[y:y+height,x:x+width]) for name, pixels in outputs.items()}
        measured = []
        for band in bands:
            witness = phase_truth[band]
            energy = float(np.vdot(witness, witness).real)
            result_band = {}
            for name, value in patch.items():
                projected = float(np.vdot(witness, value[band]).real)
                output_energy = float(np.vdot(value[band], value[band]).real)
                result_band[name] = {
                    "signed_projection": projected / energy,
                    "phase_correlation": projected / np.sqrt(energy * output_energy),
                    "amplitude_ratio": np.sqrt(output_energy / energy),
                }
            measured.append(result_band)
        result.append({"first_canvas_xywh": [x,y,width,height], "frequency_edges_cyc_per_px": octave_edges, "bands": measured})
    (folder / "spectral-diagnostics.json").write_text(json.dumps(result,indent=2,allow_nan=False)+"\n")
    for row in result:
        print(row["first_canvas_xywh"])
        for band in row["bands"]:
            print(" ".join(f"{name}={value['signed_projection']:.3f}" for name,value in band.items()))


if __name__ == "__main__":
    main()
